# 退款

API路径：`api/refund/refundOrder`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/48

## PHP 示例

```php
<?php
/**
 * 计全付 发起退款 - PHP curl 示例
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

// ========== 构造请求参数 ==========
$params = [
    'mchNo'        => $mchNo,
    'appId'        => $appId,
    'mchOrderNo'   => 'ORDER202605211430001234',      // 商户订单号（与 payOrderId 二选一）
    // 'payOrderId'   => 'P2026052114300000001',      // 或使用支付系统订单号
    'mchRefundNo'  => 'REF' . date('YmdHis') . rand(1000, 9999),  // 商户退款单号
    'refundAmount' => 1,                                // 退款金额，单位：分
    'currency'     => 'CNY',
    'refundReason' => '用户申请退款',
    'clientIp'     => '192.168.1.100',
    'notifyUrl'    => 'https://your-domain.com/notify/refund',  // 退款异步通知地址
    'version'      => '1.0',
    'signType'     => 'MD5',
    'reqTime'      => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

// ========== 发送请求 ==========
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/refund/refundOrder',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => json_encode($params),
    CURLOPT_HTTPHEADER     => [
        'Content-Type: application/json',
        'Accept: application/json',
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 30,
]);

$response = curl_exec($ch);
curl_close($ch);

$result = json_decode($response, true);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "退款发起成功\n";
    echo "refundOrderId: " . $data['refundOrderId'] . "\n";
    echo "mchRefundNo: " . $data['mchRefundNo'] . "\n";
    echo "state: " . $data['state'] . "\n";  // 0-退款生成 1-退款中 2-退款成功 3-退款失败
} else {
    echo "退款发起失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```
