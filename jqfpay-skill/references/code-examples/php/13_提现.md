# 提现

API路径：

- 提现余额查询：`api/cashout/balance/query`
- 创建提现订单：`api/cashout/order/create`
- 查询提现订单：`api/cashout/order/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/77

## PHP 示例

```php
<?php
/**
 * 计全付 提现相关接口 - PHP curl 示例
 * 包含：提现余额查询、创建提现订单、查询提现订单
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

function sendRequest(string $url, array $params): array
{
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL            => $url,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($params),
        CURLOPT_HTTPHEADER     => [
            'Content-Type: application/json',
            'Accept: application/json',
        ],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_TIMEOUT        => 30,
    ]);

    $response = curl_exec($ch);
    curl_close($ch);

    return json_decode($response, true);
}

// ==========================================================
// 1. 提现余额查询
// ==========================================================

$params = [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'ifCode'     => 'dgpay',          // 支付接口代码
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

$result = sendRequest($apiBase . '/api/cashout/balance/query', $params);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "提现余额查询成功\n";
    echo "当前余额: " . $data['balance'] . " 分\n";
    echo "冻结余额: " . $data['frozenBalance'] . " 分\n";
    echo "总余额: " . $data['totalBalance'] . " 分\n";
} else {
    echo "余额查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}

// ==========================================================
// 2. 创建提现订单
// ==========================================================

$params = [
    'mchNo'        => $mchNo,
    'appId'        => $appId,
    'mchOrderNo'   => 'CO' . date('YmdHis') . rand(1000, 9999),  // 商户提现单号
    'amount'       => 100,              // 提现金额，单位：分
    'ifCode'       => 'dgpay',          // 支付接口代码
    'notifyUrl'    => 'https://your-domain.com/notify/cashout',  // 异步通知地址
    // 'channelExtra' => '',            // 特定渠道额外参数（可选）
    'remark'       => '测试提现备注',   // 备注（可选）
    'version'      => '1.0',
    'signType'     => 'MD5',
    'reqTime'      => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

$result = sendRequest($apiBase . '/api/cashout/order/create', $params);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "提现申请成功\n";
    echo "cashoutOrderId: " . $data['cashoutOrderId'] . "\n";
    echo "mchOrderNo: " . $data['mchOrderNo'] . "\n";
    echo "state: " . $data['state'] . "\n";  // 0-提现单创建 1-提现中 2-提现成功 3-提现失败
} else {
    echo "提现申请失败: " . ($result['msg'] ?? '未知错误') . "\n";
}

// ==========================================================
// 3. 查询提现订单
// ==========================================================

$params = [
    'mchNo'          => $mchNo,
    'appId'          => $appId,
    'mchOrderNo'     => 'CO202605261530001234',       // 商户提现单号（与cashoutOrderId二选一）
    // 'cashoutOrderId' => 'C2026052600001001',       // 支付系统提现订单号
    'version'        => '1.0',
    'signType'       => 'MD5',
    'reqTime'        => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

$result = sendRequest($apiBase . '/api/cashout/order/query', $params);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "查询提现订单成功\n";
    echo "cashoutOrderId: " . $data['cashoutOrderId'] . "\n";
    echo "mchOrderNo: " . $data['mchOrderNo'] . "\n";
    echo "ifCode: " . $data['ifCode'] . "\n";
    echo "amount: " . $data['amount'] . " 分\n";
    echo "balance: " . $data['balance'] . " 分\n";
    echo "state: " . $data['state'] . "\n";
    echo "channelOrderNo: " . ($data['channelOrderNo'] ?? '') . "\n";
    echo "successTime: " . ($data['successTime'] ?? '') . "\n";
    echo "createdAt: " . ($data['createdAt'] ?? '') . "\n";
    echo "bankName: " . ($data['bankName'] ?? '') . "\n";
    echo "accountNo: " . ($data['accountNo'] ?? '') . "\n";
    echo "accountName: " . ($data['accountName'] ?? '') . "\n";
    echo "remark: " . ($data['remark'] ?? '') . "\n";
} else {
    echo "查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```

### 提现状态说明

| state | 含义 |
|-------|------|
| 0 | 提现单创建 |
| 1 | 提现中 |
| 2 | 提现成功 |
| 3 | 提现失败 |
