# 转账扩展

转账扩展接口包含：转账余额查询、转账凭证申请、转账凭证查询、资金转账支付（充值）。

API路径：
- 转账余额查询：`api/transfer/balance/query`
- 转账凭证申请：`api/transfer/certificate/apply`
- 转账凭证查询：`api/transfer/certificate/query`
- 资金转账支付：`api/fund/trans/pay`

在线文档：
- 转账余额查询：https://doc.jeequan.com/#/integrate/jqf/api/51
- 转账凭证申请：https://doc.jeequan.com/#/integrate/jqf/api/52
- 转账凭证查询：https://doc.jeequan.com/#/integrate/jqf/api/53
- 资金转账支付：https://doc.jeequan.com/#/integrate/jqf/api/54

## PHP 示例

```php
<?php
/**
 * 计全付 转账扩展 - PHP curl 示例
 * 包含：余额查询、凭证申请、凭证查询、资金转账支付
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

function sendRequest(string $apiBase, string $path, array $params, string $apiKey): array
{
    $params['sign'] = md5Sign($params, $apiKey);
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL            => $apiBase . $path,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($params),
        CURLOPT_HTTPHEADER     => [
            'Content-Type: application/json',
            'Accept: application/json',
        ],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_TIMEOUT        => 30,
    ]);
    $response = curl_exec($ch);
    curl_close($ch);
    return json_decode($response, true);
}


// ========== 1. 转账余额查询 ==========
echo "===== 转账余额查询 =====\n";
$result = sendRequest($apiBase, '/api/transfer/balance/query', [
    'mchNo'    => $mchNo,
    'appId'    => $appId,
    'ifCode'   => 'aliaqfpay',          // 支付接口代码
    'version'  => '1.0',
    'signType' => 'MD5',
    'reqTime'  => date('YmdHis'),
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    echo "可用余额: " . ($result['data']['balanceAmount'] ?? 0) . " 分\n";
} else {
    echo "查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}


// ========== 2. 转账凭证申请 ==========
echo "\n===== 转账凭证申请 =====\n";
$result = sendRequest($apiBase, '/api/transfer/certificate/apply', [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'transferId' => 'T1862403098937430017',  // 转账订单号（与 mchOrderNo 二选一）
    // 'mchOrderNo' => 'T20260521160000',    // 商户转账单号
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    echo "凭证申请成功\n";
    echo "transferId: " . ($result['data']['transferId'] ?? '') . "\n";
} else {
    echo "凭证申请失败: " . ($result['msg'] ?? '未知错误') . "\n";
}


// ========== 3. 转账凭证查询 ==========
echo "\n===== 转账凭证查询 =====\n";
$result = sendRequest($apiBase, '/api/transfer/certificate/query', [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'transferId' => 'T1862403098937430017',  // 转账订单号（与 mchOrderNo 二选一）
    'fileId'     => '2022211111',            // 文件ID
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    echo "下载地址: " . ($result['data']['downloadUrl'] ?? '') . "\n";
} else {
    echo "凭证查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}


// ========== 4. 资金转账支付（充值） ==========
echo "\n===== 资金转账支付 =====\n";
$result = sendRequest($apiBase, '/api/fund/trans/pay', [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'ifCode'      => 'aliaqfpay',       // 支付接口代码
    'amount'      => 1,                 // 充值金额，单位：分
    'rechargeType' => 0,                // 充值类型：0-线上充值 1-线下充值
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "充值类型: " . ($data['rechargeType'] ?? '') . "\n";
    if ($data['rechargeType'] == 0) {
        // 线上充值返回表单
        echo "表单内容: " . ($data['formContent'] ?? '') . "\n";
    } else {
        // 线下充值返回银行卡信息
        echo "卡号: " . ($data['cardNo'] ?? '') . "\n";
        echo "开户行: " . ($data['cardBank'] ?? '') . "\n";
    }
} else {
    echo "充值失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```
