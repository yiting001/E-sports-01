# 分账

API路径：`api/division/exec`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/57

## PHP 示例

```php
<?php
/**
 * 计全付 发起分账 - PHP curl 示例
 *
 * 前置条件：
 * 1. 统一下单时 divisionMode 设置为 2（手动分账），订单支付成功后资金冻结
 * 2. 已通过 api/division/receiver/bind 绑定分账接收方
 * 3. 调用本接口发起分账
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

// ========== 构造请求参数 ==========
$params = [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'mchOrderNo' => 'ORDER202605211430001234',  // 商户订单号（与 payOrderId 二选一）
    // 'payOrderId' => 'P2026052114300000001',  // 或使用支付系统订单号

    // 方式1：使用系统自动分账组
    'useSysAutoDivisionReceivers' => 1,  // 1-使用系统配置的分账组 0-不使用

    // 方式2：手动指定分账接收方（与 useSysAutoDivisionReceivers 二选一）
    // 'receivers' => json_encode([
    //     [
    //         'receiverId'     => 800001,            // 分账接收方ID（绑定分账时获得）
    //         'divisionProfit' => 0.1,               // 分账比例，0.1 = 10%
    //     ],
    //     [
    //         'receiverGroupId' => 100001,           // 分账接收方组ID
    //         'divisionProfit'  => 0.15,             // 每个账号的分账比例
    //     ],
    // ]),

    // 是否自动完成分账（解冻剩余资金）
    'autoFinish' => 1,  // 0-否 1-是（分账完成后自动解冻剩余未分资金）

    // 分账模板（可选）
    // 'templateId' => 200001,

    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

// ========== 发送请求 ==========
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/division/exec',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => json_encode($params),
    CURLOPT_HTTPHEADER     => [
        'Content-Type: application/json',
        'Accept: application/json',
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 30,
]);

$response = curl_exec($ch);
curl_close($ch);

$result = json_decode($response, true);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "分账发起成功\n";
    echo "batchOrderId: " . ($data['batchOrderId'] ?? '') . "\n";
    echo "state: " . ($data['state'] ?? '') . "\n";  // 0-分账生成 1-分账中 2-分账成功 3-分账失败
} else {
    echo "分账失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```
