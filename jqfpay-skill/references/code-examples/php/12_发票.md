# 发票

API路径：

- 开票：`api/invoice/open`
- 查询发票：`api/invoice/query`
- 电子发票URL：`api/invoice/elecUrl`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/74

## PHP 示例

```php
<?php
/**
 * 计全付 发票相关接口 - PHP curl 示例
 * 包含：开票、查询发票、获取电子发票URL
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

function sendRequest(string $url, array $params): array
{
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL            => $url,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($params),
        CURLOPT_HTTPHEADER     => [
            'Content-Type: application/json',
            'Accept: application/json',
        ],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_TIMEOUT        => 30,
    ]);

    $response = curl_exec($ch);
    curl_close($ch);

    return json_decode($response, true);
}

// ==========================================================
// 1. 开票
// ==========================================================

$buyerInfo = json_encode([
    'buyerName'  => '测试公司',
    'taxNo'      => '91110108MA01XXXXXX',
    'pushPhone'  => '13800138000',
]);

$params = [
    'mchNo'        => $mchNo,
    'appId'        => $appId,
    'mchOrderNo'   => 'FP' . date('YmdHis') . rand(1000, 9999),  // 商户开票单号
    'invoiceWay'   => 2,                // 开票方式: 2-获取开票链接
    'amount'       => 100,              // 含税开票金额，单位：分
    'clientIp'     => '127.0.0.1',      // 客户端IP地址
    'buyerInfo'    => $buyerInfo,       // 购方信息
    'notifyUrl'    => 'https://your-domain.com/notify/invoice',  // 异步通知地址
    // 'expiredDay'   => 7,             // 开票链接有效期（可选）
    // 'remark'       => '备注信息',    // 备注（可选）
    'version'      => '1.0',
    'signType'     => 'MD5',
    'reqTime'      => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

$result = sendRequest($apiBase . '/api/invoice/open', $params);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "开票申请成功\n";
    echo "invoiceOrderId: " . $data['invoiceOrderId'] . "\n";
    echo "state: " . $data['state'] . "\n";  // 0-订单生成 1-开票中 2-开票成功 3-开票失败 4-开票任务关闭 5-已冲红 6-已作废
    echo "channelInvoiceQrcodeUrl: " . $data['channelInvoiceQrcodeUrl'] . "\n";
    echo "expiredTime: " . $data['expiredTime'] . "\n";
} else {
    echo "开票失败: " . ($result['msg'] ?? '未知错误') . "\n";
}

// ==========================================================
// 2. 查询发票
// ==========================================================

$params = [
    'mchNo'            => $mchNo,
    'appId'            => $appId,
    'invoiceOrderId'   => 'FP2026052600001001',   // 发票单号（与mchOrderNo二选一）
    // 'mchOrderNo'    => 'FP202605261530001234',  // 商户开票单号
    'version'          => '1.0',
    'signType'         => 'MD5',
    'reqTime'          => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

$result = sendRequest($apiBase . '/api/invoice/query', $params);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "查询发票成功\n";
    echo "invoiceOrderId: " . $data['invoiceOrderId'] . "\n";
    echo "mchOrderNo: " . $data['mchOrderNo'] . "\n";
    echo "amount: " . $data['amount'] . " 分\n";
    echo "feeAmount: " . $data['feeAmount'] . " 分\n";
    echo "state: " . $data['state'] . "\n";
    echo "channelInvoiceOrderNo: " . ($data['channelInvoiceOrderNo'] ?? '') . "\n";
} else {
    echo "查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}

// ==========================================================
// 3. 获取电子发票URL（全电账号管理地址）
// ==========================================================

$params = [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

$result = sendRequest($apiBase . '/api/invoice/elecUrl', $params);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "获取电子发票URL成功\n";
    echo "etaxUserUrl: " . $data['etaxUserUrl'] . "\n";
} else {
    echo "获取失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```

### 开票状态说明

| state | 含义 |
|-------|------|
| 0 | 订单生成 |
| 1 | 开票中 |
| 2 | 开票成功 |
| 3 | 开票失败 |
| 4 | 开票任务关闭 |
| 5 | 已冲红 |
| 6 | 已作废 |
