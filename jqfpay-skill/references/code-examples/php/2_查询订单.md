# 查询订单

API路径：`api/pay/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/40

## PHP 示例

```php
<?php
/**
 * 计全付 查询支付订单 - PHP curl 示例
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

// ========== 构造请求参数 ==========
// 使用商户订单号查询（与 payOrderId 二选一）
$params = [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'mchOrderNo' => 'ORDER202605211430001234',  // 商户订单号
    // 'payOrderId' => 'P2026052114300000001',  // 或使用支付系统订单号
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

// ========== 发送请求 ==========
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/pay/query',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => json_encode($params),
    CURLOPT_HTTPHEADER     => [
        'Content-Type: application/json',
        'Accept: application/json',
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 30,
]);

$response = curl_exec($ch);
curl_close($ch);

$result = json_decode($response, true);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "payOrderId: " . $data['payOrderId'] . "\n";
    echo "mchOrderNo: " . $data['mchOrderNo'] . "\n";
    echo "orderState: " . $data['state'] . "\n";     // 0-订单生成 1-支付中 2-支付成功 3-支付失败 4-已撤销 5-已退款 6-退款中
    echo "amount: " . $data['amount'] . "\n";
    echo "wayCode: " . $data['wayCode'] . "\n";
    echo "successTime: " . ($data['successTime'] ?? '') . "\n";
} else {
    echo "查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```
