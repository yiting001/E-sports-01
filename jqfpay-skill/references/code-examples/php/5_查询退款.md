# 查询退款

API路径：`api/refund/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/49

## PHP 示例

```php
<?php
/**
 * 计全付 查询退款 - PHP curl 示例
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

// ========== 构造请求参数 ==========
// 支持三种查询维度（任选其一）：
// 1. mchRefundNo  - 商户退款单号
// 2. refundOrderId - 退款系统订单号
// 3. mchOrderNo + refundAmount - 商户订单号 + 退款金额
$params = [
    'mchNo'        => $mchNo,
    'appId'        => $appId,
    'mchRefundNo'  => 'REF202605211500001234',  // 商户退款单号
    // 'refundOrderId' => 'RF2026052115000000001',  // 或退款系统订单号
    'version'      => '1.0',
    'signType'     => 'MD5',
    'reqTime'      => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

// ========== 发送请求 ==========
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/refund/query',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => json_encode($params),
    CURLOPT_HTTPHEADER     => [
        'Content-Type: application/json',
        'Accept: application/json',
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 30,
]);

$response = curl_exec($ch);
curl_close($ch);

$result = json_decode($response, true);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "refundOrderId: " . $data['refundOrderId'] . "\n";
    echo "state: " . $data['state'] . "\n";  // 0-退款生成 1-退款中 2-退款成功 3-退款失败
    echo "refundAmount: " . $data['refundAmount'] . "\n";
    echo "channelOrderNo: " . ($data['channelOrderNo'] ?? '') . "\n";
    echo "successTime: " . ($data['successTime'] ?? '') . "\n";
} else {
    echo "查询失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```
