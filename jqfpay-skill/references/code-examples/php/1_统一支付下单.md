# 统一支付下单

API路径：`api/pay/unifiedOrder`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/39

## PHP 示例

```php
<?php
/**
 * 计全付 统一支付下单 - PHP curl 示例（手动 MD5 签名）
 */

// ========== 配置 ==========
$apiBase = 'https://your-jqf-domain.com';  // 计全付服务地址
$appId   = '6064c5c8f3e4a24e5b8e1a3c';     // 商户应用ID
$mchNo   = 'M1640000001';                   // 商户号
$apiKey  = 'your_api_key_here';             // 商户应用密钥（AppSecret）

// ========== 签名函数 ==========
/**
 * MD5 签名
 * 1. 收集所有非空参数
 * 2. 按 key 的 ASCII 字典序排序（不区分大小写）
 * 3. 拼接 key1=value1&key2=value2... 格式
 * 4. 末尾追加 &key=apiKey
 * 5. MD5 哈希后转大写
 */
function md5Sign(array $params, string $apiKey): string
{
    // 过滤空值
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });

    // 按 key 不区分大小写排序
    ksort($filtered, SORT_STRING | SORT_FLAG);

    // 拼接待签名字符串
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;

    return strtoupper(md5($signStr));
}

// ========== 构造请求参数 ==========
$params = [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'mchOrderNo' => 'ORDER' . date('YmdHis') . rand(1000, 9999),
    'wayCode'    => 'WX_JSAPI',                    // 支付方式：微信公众号
    'amount'     => 1,                              // 金额，单位：分
    'currency'   => 'CNY',
    'clientIp'   => '192.168.1.100',
    'subject'    => '测试商品标题',
    'body'       => '测试商品描述',
    'notifyUrl'  => 'https://your-domain.com/notify/pay',
    'returnUrl'  => 'https://your-domain.com/pay/return',
    'channelExtra' => json_encode([                 // 渠道额外参数
        'openid' => 'o6BcIwvSiRpfS8e_UyfQNrYuk2LI'  // 微信JSAPI需传openid
    ]),
    'divisionMode' => 0,                            // 0-不允许分账
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
];

// 计算签名
$params['sign'] = md5Sign($params, $apiKey);

// ========== 发送请求 ==========
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/pay/unifiedOrder',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => json_encode($params),
    CURLOPT_HTTPHEADER     => [
        'Content-Type: application/json',
        'Accept: application/json',
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 30,
]);

$response = curl_exec($ch);
$httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
$curlError = curl_error($ch);
curl_close($ch);

if ($curlError) {
    die('CURL Error: ' . $curlError);
}

// ========== 解析响应 ==========
$result = json_decode($response, true);
echo "HTTP Status: {$httpCode}\n";
echo "Response Code: {$result['code']}\n";
echo "Response Msg: {$result['msg']}\n";

if ($result['code'] === 0 || $result['code'] === 1) {
    // 成功
    $data = $result['data'] ?? [];
    echo "payOrderId: " . ($data['payOrderId'] ?? '') . "\n";
    echo "payDataType: " . ($data['payDataType'] ?? '') . "\n";
    echo "payData: " . ($data['payData'] ?? '') . "\n";

    // 根据支付参数类型处理
    // payDataType = wxapp     → 微信支付参数，调用微信SDK支付
    // payDataType = codeUrl   → 二维码链接，前端生成二维码
    // payDataType = payurl    → 跳转链接，重定向
    // payDataType = form      → 自动提交表单
    // payDataType = none      → 无需支付参数（条码直接完成）
} else {
    echo "下单失败: " . $result['msg'] . "\n";
}

// ========== 不同支付方式示例 ==========

// --- WX_LITE 微信小程序 ---
// 'wayCode' => 'WX_LITE',
// 'channelExtra' => json_encode(['openid' => '用户的openid']),

// --- QR_CASHIER 聚合扫码（用户扫商家）---
// 'wayCode' => 'QR_CASHIER',
// 'channelExtra' => json_encode(['payDataType' => 'codeImgUrl']),

// --- AUTO_BAR 聚合条码（商家扫用户）---
// 'wayCode' => 'AUTO_BAR',
// 'channelExtra' => json_encode(['authCode' => '用户付款码']),

// --- ALI_APP 支付宝APP ---
// 'wayCode' => 'ALI_APP',
// 'channelExtra' => '{}',

// --- WEB_CASHIER WEB收银台 ---
// 'wayCode' => 'WEB_CASHIER',
// 'channelExtra' => '{}',
```
