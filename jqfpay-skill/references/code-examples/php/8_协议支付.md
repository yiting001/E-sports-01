# 协议支付

API路径：
- 预绑定：`api/pay/bankAgreePreBind`
- 确认绑定：`api/pay/bankAgreeConfirmBind`
- 协议支付确认：`api/pay/bankAgreeProtocolpayConfirm`
- 重发支付短信：`api/pay/bankAgreeResendPaysms`
- 解绑协议：`api/pay/bankAgreeUnBind`
- 银行列表查询：`api/pay/bankAgreeBanklistQuery`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/131

## PHP 示例

```php
<?php
/**
 * 计全付 协议支付（快捷支付-协议模式）- PHP curl 完整流程示例
 *
 * 完整流程：
 * 1. 银行列表查询（可选）→ 获取支持的银行
 * 2. 协议预绑定 → 提交银行卡信息
 * 3. 协议确认绑定 → 用户输入短信验证码确认绑定
 * 4. 协议支付确认 → 发起扣款（可重复使用已绑定的协议）
 * 5. 重发支付短信（可选）→ 重新获取验证码
 * 6. 解绑协议（可选）→ 解除绑定
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

function postJson(string $url, array $params, string $apiKey): array
{
    $params['sign'] = md5Sign($params, $apiKey);
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL            => $url,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($params),
        CURLOPT_HTTPHEADER     => [
            'Content-Type: application/json',
            'Accept: application/json',
        ],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_TIMEOUT        => 30,
    ]);
    $response = curl_exec($ch);
    curl_close($ch);
    return json_decode($response, true) ?: [];
}

// =================================================================
// Step 1: 银行列表查询（可选）
// =================================================================
echo "===== Step 1: 查询支持的银行列表 =====\n";
$result = postJson($apiBase . '/api/pay/bankAgreeBanklistQuery', [
    'mchNo'    => $mchNo,
    'appId'    => $appId,
    'version'  => '1.0',
    'signType' => 'MD5',
    'reqTime'  => date('YmdHis'),
], $apiKey);
print_r($result);

// =================================================================
// Step 2: 协议预绑定 - 提交银行卡信息，获取短信验证码
// =================================================================
echo "\n===== Step 2: 协议预绑定 =====\n";
$bindParams = [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'mchOrderNo'  => 'AGREE_PRE' . date('YmdHis') . rand(1000, 9999),
    'bankCardNo'  => '6222021234567890123',   // 银行卡号
    'bankCardName'=> '张三',                   // 持卡人姓名
    'certNo'      => '110101199001011234',     // 身份证号
    'phoneNo'     => '13800138000',            // 银行预留手机号
    'bankCode'    => 'ICBC',                   // 银行编码（从银行列表获取）
    'bankAccType' => '1',                      // 账户类型 1-借记卡 2-信用卡
    'notifyUrl'   => 'https://your-domain.com/notify/agree',
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
];
$result = postJson($apiBase . '/api/pay/bankAgreePreBind', $bindParams, $apiKey);
$agreementId = $result['data']['agreementId'] ?? '';  // 保存协议ID，后续步骤需要
echo "agreementId: {$agreementId}\n";
print_r($result);

// =================================================================
// Step 3: 协议确认绑定 - 用户输入短信验证码确认
// =================================================================
echo "\n===== Step 3: 协议确认绑定 =====\n";
$smsCode = '123456';  // 用户收到的短信验证码
$result = postJson($apiBase . '/api/pay/bankAgreeConfirmBind', [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'agreementId' => $agreementId,              // Step 2 获取的协议ID
    'smsCode'     => $smsCode,                  // 短信验证码
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
], $apiKey);
print_r($result);

// =================================================================
// Step 4: 协议支付确认 - 使用已绑定的协议发起扣款
// =================================================================
echo "\n===== Step 4: 协议支付确认 =====\n";
$payResult = postJson($apiBase . '/api/pay/bankAgreeProtocolpayConfirm', [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'mchOrderNo'  => 'AGREE_PAY' . date('YmdHis') . rand(1000, 9999),  // 支付订单号
    'agreementId' => $agreementId,              // 使用已绑定的协议ID
    'amount'      => 100,                       // 支付金额，单位：分
    'currency'    => 'CNY',
    'subject'     => '协议支付测试',
    'body'        => '协议支付商品描述',
    'notifyUrl'   => 'https://your-domain.com/notify/pay',
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
], $apiKey);
print_r($payResult);
// 返回结果：
//   code=0, state=1(支付中) → 需要输入短信验证码 → 调用 Step 4b
//   code=0, state=2(支付成功) → 支付完成

// Step 4b: 如支付需要短信验证码（确认支付）
if (($payResult['data']['state'] ?? 0) === 1) {
    echo "\n===== Step 4b: 输入短信验证码确认支付 =====\n";
    $paySmsCode = '654321';
    $result = postJson($apiBase . '/api/pay/bankAgreeProtocolpayConfirm', [
        'mchNo'          => $mchNo,
        'appId'          => $appId,
        'agreementId'    => $agreementId,
        'mchOrderNo'     => $payResult['data']['mchOrderNo'] ?? '',
        'payOrderId'     => $payResult['data']['payOrderId'] ?? '',
        'smsCode'        => $paySmsCode,
        'amount'         => 100,
        'currency'       => 'CNY',
        'subject'        => '协议支付测试',
        'body'           => '协议支付商品描述',
        'notifyUrl'      => 'https://your-domain.com/notify/pay',
        'version'        => '1.0',
        'signType'       => 'MD5',
        'reqTime'        => date('YmdHis'),
    ], $apiKey);
    print_r($result);
}

// =================================================================
// Step 5: 重发支付短信（可选）
// =================================================================
echo "\n===== Step 5: 重发支付短信 =====\n";
$result = postJson($apiBase . '/api/pay/bankAgreeResendPaysms', [
    'mchNo'      => $mchNo,
    'appId'      => $appId,
    'agreementId' => $agreementId,
    'mchOrderNo' => 'AGREE_PAY202605211600001234',
    'version'    => '1.0',
    'signType'   => 'MD5',
    'reqTime'    => date('YmdHis'),
], $apiKey);
print_r($result);

// =================================================================
// Step 6: 解绑协议（可选）
// =================================================================
echo "\n===== Step 6: 解绑协议 =====\n";
$result = postJson($apiBase . '/api/pay/bankAgreeUnBind', [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'agreementId' => $agreementId,
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
], $apiKey);
print_r($result);
```
