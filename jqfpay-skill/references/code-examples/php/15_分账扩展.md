# 分账扩展

分账扩展接口包含：绑定/查询分账用户、查询/完结分账、余额分账执行/查询、渠道余额查询/提现、提现记录查询、结算更新。

API路径：
- 绑定分账用户：`api/division/receiver/bind`
- 查询分账绑定：`api/division/receiver/bindQuery`
- 查询分账：`api/division/query`
- 完成分账：`api/division/finish`
- 余额分账执行：`api/division/balance/exec`
- 余额分账查询：`api/division/balance/query`
- 渠道余额查询：`api/division/receiver/channelBalanceQuery`
- 渠道余额提现：`api/division/receiver/channelBalanceCashout`
- 提现记录查询：`api/division/receiver/cashoutRecordQuery`
- 结算更新：`api/division/receiver/settle/update`

## PHP 示例

```php
<?php
/**
 * 计全付 分账扩展 - PHP curl 示例
 * 包含：绑定/查询用户、查询/完结分账、余额分账、渠道余额、提现记录、结算更新
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

function sendRequest(string $apiBase, string $path, array $params, string $apiKey): array
{
    $params['mchNo']    = $params['mchNo'] ?? $GLOBALS['mchNo'];
    $params['appId']    = $params['appId'] ?? $GLOBALS['appId'];
    $params['version']  = $params['version'] ?? '1.0';
    $params['signType'] = $params['signType'] ?? 'MD5';
    $params['reqTime']  = $params['reqTime'] ?? date('YmdHis');
    $params['sign']     = md5Sign($params, $apiKey);

    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL            => $apiBase . $path,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($params),
        CURLOPT_HTTPHEADER     => [
            'Content-Type: application/json',
            'Accept: application/json',
        ],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_TIMEOUT        => 30,
    ]);
    $response = curl_exec($ch);
    curl_close($ch);
    return json_decode($response, true);
}


// ========== 1. 绑定分账用户 ==========
echo "===== 绑定分账用户 =====\n";
$result = sendRequest($apiBase, '/api/division/receiver/bind', [
    'ifCode'          => 'demomockpay',            // 接口代码
    'mchReceiverId'   => '23472389482349',         // 商户分账用户ID
    'receiverAlias'   => '15521254124',            // 接收者账号别名
    'accType'         => 0,                        // 账号类型：0-个人 1-商户
    'accNo'           => '15521254124',            // 分账接收账号
    'relationType'    => 'PARTNER',                // 分账关系类型
    'divisionProfit'  => '0.3',                    // 默认分账比例
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "绑定成功 receiverId: " . ($data['receiverId'] ?? '') . "\n";
    echo "绑定状态: " . ($data['bindState'] ?? '') . "\n";
} else {
    echo "绑定失败: " . ($result['msg'] ?? '未知错误') . "\n";
}


// ========== 2. 查询分账绑定 ==========
echo "\n===== 查询分账绑定 =====\n";
$result = sendRequest($apiBase, '/api/division/receiver/bindQuery', [
    'receiverId' => 800093,                    // 分账接收者ID（与 mchReceiverId 二选一）
    // 'mchReceiverId' => '23472389482349',
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "绑定状态: " . ($data['bindState'] ?? '') . "\n";
    echo "分账比例: " . ($data['divisionProfit'] ?? '') . "\n";
}


// ========== 3. 查询分账 ==========
echo "\n===== 查询分账 =====\n";
$result = sendRequest($apiBase, '/api/division/query', [
    'payOrderId'   => 'P1955471412902871041',     // 支付订单号（与 mchOrderNo 二选一）
    'batchOrderId' => 'D1955471507933216769',     // 系统分账批次号
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "分账状态: " . ($data['divisionState'] ?? '') . "\n";  // 0-未分账 1-待分账 2-分账中 3-已分账 4-完结中 5-已完结
}


// ========== 4. 完成分账 ==========
echo "\n===== 完成分账 =====\n";
$result = sendRequest($apiBase, '/api/division/finish', [
    'payOrderId' => 'P1955471412902871041',     // 支付订单号（与 mchOrderNo 二选一）
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "完结状态: " . ($data['state'] ?? '') . "\n";  // 1-成功 2-失败 3-处理中
}


// ========== 5. 余额分账执行 ==========
echo "\n===== 余额分账执行 =====\n";
$receivers = json_encode([
    ['receiverId' => 800085, 'divisionAmount' => '1']
]);
$result = sendRequest($apiBase, '/api/division/balance/exec', [
    'mchDivisionNo' => 'mho' . round(microtime(true) * 1000),  // 商户分账单号
    'ifCode'        => 'demomockpay',                            // 支付接口代码
    'totalAmount'   => 1,                                       // 分账总金额，单位：分
    'receivers'     => $receivers,                               // 分账接收者列表
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "分账状态: " . ($data['state'] ?? '') . "\n";  // 1-成功 2-失败 3-待分账 4-分账中
    echo "系统批次号: " . ($data['batchOrderId'] ?? '') . "\n";
}


// ========== 6. 余额分账查询 ==========
echo "\n===== 余额分账查询 =====\n";
$result = sendRequest($apiBase, '/api/division/balance/query', [
    'mchDivisionNo' => 'mho1716288000000',      // 商户分账单号（与 batchOrderId 二选一）
    // 'batchOrderId' => 'D202605211700000001',
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "系统批次号: " . ($data['batchOrderId'] ?? '') . "\n";
}


// ========== 7. 渠道余额查询 ==========
echo "\n===== 渠道余额查询 =====\n";
$result = sendRequest($apiBase, '/api/division/receiver/channelBalanceQuery', [
    'receiverId' => 800057,                    // 分账用户ID
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "可提现余额: " . ($data['balanceAmount'] ?? 0) . " 分\n";
    echo "不可提现余额: " . ($data['unBalanceAmount'] ?? 0) . " 分\n";
}


// ========== 8. 渠道余额提现 ==========
echo "\n===== 渠道余额提现 =====\n";
$result = sendRequest($apiBase, '/api/division/receiver/channelBalanceCashout', [
    'receiverId'    => 800057,                                    // 分账用户ID
    'mchOrderNo'    => 'CO' . date('YmdHis') . rand(1000, 9999), // 商户提现单号
    'cashoutAmount' => 100,                                       // 提现金额，单位：分
    'remark'        => '分账提现',
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "提现记录单号: " . ($data['rid'] ?? '') . "\n";
    echo "发起状态: " . ($data['state'] ?? '') . "\n";           // 1-成功 0-失败
    echo "提现状态: " . ($data['cashoutState'] ?? '') . "\n";    // 0-创建 1-提现中 2-成功 3-失败
}


// ========== 9. 提现记录查询 ==========
echo "\n===== 提现记录查询 =====\n";
$result = sendRequest($apiBase, '/api/division/receiver/cashoutRecordQuery', [
    'receiverId' => 800057,
    'pageNumber' => 1,                         // 分页页码
    'pageSize'   => 20,                        // 分页条数
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "总条数: " . ($data['total'] ?? 0) . "\n";
    echo "是否有下一页: " . ($data['hasNext'] ? 'true' : 'false') . "\n";
}


// ========== 10. 结算更新 ==========
echo "\n===== 结算更新 =====\n";
$result = sendRequest($apiBase, '/api/division/receiver/settle/update', [
    'receiverId'                 => 800046,
    'settAccountName'            => '测试',
    'settAccountNo'              => '6217375892834829482',
    'settAccountBankName'        => '中国建设银行',
    'settAccountBankBranchCode'  => '105143000017',
    'settAccountBankBranchName'  => '',
    'channelExtInfo'             => '',
], $apiKey);

if (($result['code'] ?? -1) === 0) {
    echo "结算信息更新成功\n";
}
```
