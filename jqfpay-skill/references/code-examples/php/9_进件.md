# 进件

API路径：
- 上传文件：`api/mchApplyment/uploadFile`
- 商户进件：`api/mchApplyment/income`
- 查询进件：`api/mchApplyment/income/query`
- 签约信息：`api/mchApplyment/signInfo`

在线文档：
- 上传文件：https://doc.jeequan.com/#/integrate/jqf/api/344
- 商户进件：https://doc.jeequan.com/#/integrate/jqf/api/343

## PHP 示例

```php
<?php
/**
 * 计全付 商户进件 - PHP curl 示例
 *
 * 进件流程：
 * 1. 上传证件图片文件 → 获取 fileId
 * 2. 提交进件申请 → 获取 applymentId
 * 3. 轮询查询进件状态 或 等待异步通知
 * 4. 签约 → 获取签约URL
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

function postJson(string $url, array $params, string $apiKey): array
{
    $params['sign'] = md5Sign($params, $apiKey);
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL            => $url,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($params),
        CURLOPT_HTTPHEADER     => [
            'Content-Type: application/json',
            'Accept: application/json',
        ],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_TIMEOUT        => 30,
    ]);
    $response = curl_exec($ch);
    curl_close($ch);
    return json_decode($response, true) ?: [];
}

// =================================================================
// Step 1: 上传文件（multipart/form-data）
// =================================================================
echo "===== Step 1: 上传证件图片 =====\n";

// 构造签名参数（不含文件内容）
$signParams = [
    'mchNo'    => $mchNo,
    'appId'    => $appId,
    'version'  => '1.0',
    'signType' => 'MD5',
    'reqTime'  => date('YmdHis'),
];
$signParams['sign'] = md5Sign($signParams, $apiKey);

// 使用 multipart/form-data 上传文件
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/mchApplyment/uploadFile',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => [
        'mchNo'    => $mchNo,
        'appId'    => $appId,
        'version'  => '1.0',
        'signType' => 'MD5',
        'reqTime'  => $signParams['reqTime'],
        'sign'     => $signParams['sign'],
        'file'     => new CURLFile('/path/to/id_card_front.jpg', 'image/jpeg', 'id_card_front.jpg'),
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 60,
]);

$response = curl_exec($ch);
curl_close($ch);

$uploadResult = json_decode($response, true) ?: [];
$fileId = $uploadResult['data']['fileId'] ?? '';
echo "fileId: {$fileId}\n";

// 多次上传不同证件（身份证正面、反面、手持、营业执照等）
// $idCardFrontFileId = ...;
// $idCardBackFileId = ...;
// $businessLicenseFileId = ...;

// =================================================================
// Step 2: 提交进件申请（JSON格式）
// =================================================================
echo "\n===== Step 2: 提交进件申请 =====\n";

$applymentParams = [
    'mchNo'    => $mchNo,
    'appId'    => $appId,
    'version'  => '1.0',
    'signType' => 'MD5',
    'reqTime'  => date('YmdHis'),

    // 商户基本信息
    'mchShortName'   => '测试商户',
    'mchName'        => '北京测试科技有限公司',
    'mchType'        => '1',                    // 1-个体户 2-企业
    'licenseNo'      => '91110108MA01ABCDEF',   // 营业执照号
    'licenseFileId'  => $fileId,                // Step 1 上传的营业执照文件ID
    'idType'         => '1',                    // 证件类型 1-身份证
    'idCardNo'       => '110101199001011234',   // 法人身份证号
    'idCardFrontFileId' => $fileId,             // 身份证正面文件ID
    'idCardBackFileId'  => $fileId,             // 身份证反面文件ID
    'contactName'    => '张三',
    'contactPhone'   => '13800138000',
    'contactEmail'   => 'zhangsan@example.com',

    // 结算信息
    'settleAcctType'    => '1',                 // 1-对私 2-对公
    'settleAcctBank'    => '工商银行',
    'settleAcctBankNo'  => '6222021234567890123',
    'settleAcctName'    => '张三',

    // 经营信息
    'provinceCode'   => '110000',
    'cityCode'       => '110100',
    'districtCode'   => '110108',
    'address'        => '北京市海淀区xxx路xxx号',

    // 渠道信息（需要开通的支付渠道配置）
    'channelExtInfo' => json_encode([
        'wxpay' => [
            'appType' => 'JSAPI',
        ],
    ]),

    'notifyUrl' => 'https://your-domain.com/notify/applyment',
];

$applyResult = postJson($apiBase . '/api/mchApplyment/income', $applymentParams, $apiKey);
$applymentId = $applyResult['data']['applymentId'] ?? '';
echo "applymentId: {$applymentId}\n";
print_r($applyResult);

// =================================================================
// Step 3: 查询进件状态
// =================================================================
echo "\n===== Step 3: 查询进件状态 =====\n";
$queryResult = postJson($apiBase . '/api/mchApplyment/income/query', [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'applymentId' => $applymentId,
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
], $apiKey);
// state: 0-审核中 1-审核通过 2-审核驳回 3-待签约
echo "state: " . ($queryResult['data']['state'] ?? '') . "\n";
print_r($queryResult);

// =================================================================
// Step 4: 获取签约信息
// =================================================================
echo "\n===== Step 4: 获取签约URL =====\n";
$signResult = postJson($apiBase . '/api/mchApplyment/signInfo', [
    'mchNo'       => $mchNo,
    'appId'       => $appId,
    'applymentId' => $applymentId,
    'version'     => '1.0',
    'signType'    => 'MD5',
    'reqTime'     => date('YmdHis'),
], $apiKey);
echo "signUrl: " . ($signResult['data']['signUrl'] ?? '') . "\n";
print_r($signResult);
```
