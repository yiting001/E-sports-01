# 异步通知

API路径：无（计全付主动推送）

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/44

## PHP 示例

```php
<?php
/**
 * 计全付 异步通知处理 - PHP 示例
 *
 * 计全付会在支付、退款、转账等操作完成后，向商户配置的 notifyUrl 推送异步通知。
 * 商户需要：
 * 1. 验证签名（防止伪造通知）
 * 2. 处理业务逻辑（更新订单状态等）
 * 3. 返回 "SUCCESS" 字符串（计全付收到后停止重推）
 *
 * 注意：
 * - 通知可能重复推送，商户需做幂等处理
 * - 验签时需要排除 sign 字段
 * - 必须在 5 秒内返回响应
 */

$apiKey = 'your_api_key_here';

/**
 * 验证签名
 * @param array $params 通知参数（包含 sign 字段）
 * @param string $apiKey 商户应用密钥
 * @return bool 验签是否通过
 */
function verifySign(array $params, string $apiKey): bool
{
    $sign = $params['sign'] ?? '';
    if (empty($sign)) {
        return false;
    }

    // 移除 sign 字段
    unset($params['sign']);

    // 过滤空值
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });

    // 按 key 不区分大小写排序
    ksort($filtered, SORT_STRING | SORT_FLAG);

    // 拼接待签名字符串
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;

    // 计算签名并比对
    $calculatedSign = strtoupper(md5($signStr));

    return $sign === $calculatedSign;
}

// =================================================================
// 处理支付结果通知
// =================================================================

// 获取计全付推送的通知数据（JSON格式，读取请求体）
$rawInput = file_get_contents('php://input');

// 如果计全付以 form 表单形式推送（部分场景）
if (empty($notifyData)) {
    $notifyData = $_POST;
}

if (empty($notifyData)) {
    echo 'FAIL';
    exit;
}

// 记录日志（建议）
error_log('[JQF Notify] Raw: ' . $rawInput);

// 验证签名
if (!verifySign($notifyData, $apiKey)) {
    error_log('[JQF Notify] Sign verification failed');
    echo 'FAIL';
    exit;
}

// 解析通知参数
$payOrderId   = $notifyData['payOrderId'] ?? '';    // 支付订单号
$mchOrderNo   = $notifyData['mchOrderNo'] ?? '';    // 商户订单号
$state        = $notifyData['state'] ?? '';          // 订单状态：2-支付成功 3-支付失败
$amount       = $notifyData['amount'] ?? 0;          // 支付金额
$wayCode      = $notifyData['wayCode'] ?? '';        // 支付方式
$successTime  = $notifyData['successTime'] ?? '';    // 支付成功时间

// 根据订单状态处理业务
if ($state == 2) {
    // === 支付成功 ===

    // 1. 查询本地订单
    // $order = db_query("SELECT * FROM orders WHERE mch_order_no = ?", [$mchOrderNo]);

    // 2. 幂等校验：如果订单已经处理过，直接返回成功

    //     echo 'SUCCESS';

    // }

    // 3. 校验金额是否一致
    // if ($order['amount'] != $amount) {
    //     error_log('[JQF Notify] Amount mismatch');
    //     echo 'FAIL';
    //     exit;
    // }

    // 4. 更新订单状态
    // db_update("UPDATE orders SET status = 'PAID', pay_time = ? WHERE mch_order_no = ?", [date('Y-m-d H:i:s'), $mchOrderNo]);

    // 5. 其他业务逻辑（发货、发通知等）
    error_log("[JQF Notify] Payment success: {$mchOrderNo}, amount: {$amount}");

} elseif ($state == 3) {
    // === 支付失败 ===
    // 更新订单状态为失败
    error_log("[JQF Notify] Payment failed: {$mchOrderNo}");
}

// 返回成功标识（必须返回 "SUCCESS"，否则计全付会持续重推）
echo 'SUCCESS';
```
