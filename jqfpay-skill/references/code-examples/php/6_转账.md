# 转账

API路径：`api/transferOrder`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/45

## PHP 示例

```php
<?php
/**
 * 计全付 发起转账 - PHP curl 示例
 */

$apiBase = 'https://your-jqf-domain.com';
$appId   = '6064c5c8f3e4a24e5b8e1a3c';
$mchNo   = 'M1640000001';
$apiKey  = 'your_api_key_here';

function md5Sign(array $params, string $apiKey): string
{
    $filtered = array_filter($params, function ($v) {
        return $v !== null && $v !== '';
    });
    ksort($filtered, SORT_STRING | SORT_FLAG);
    $parts = [];
    foreach ($filtered as $k => $v) {
        $parts[] = $k . '=' . $v;
    }
    $signStr = implode('&', $parts) . '&key=' . $apiKey;
    return strtoupper(md5($signStr));
}

// ========== 构造请求参数 ==========
$params = [
    'mchNo'        => $mchNo,
    'appId'        => $appId,
    'mchOrderNo'   => 'TRANS' . date('YmdHis') . rand(1000, 9999),  // 商户转账单号
    'ifCode'       => 'wxpay',          // 支付接口代码（如 wxpay, alipay 等）
    'entryType'    => 'WX_CASH',        // 入账方式：WX_CASH-微信零钱 ALI_BALANCE-支付宝余额 BANK_CARD-银行卡
    'amount'       => 100,              // 转账金额，单位：分
    'currency'     => 'CNY',
    'accountNo'    => 'o6BcIwvSiRpfS8e_UyfQNrYuk2LI',  // 收款账号（微信openid / 支付宝userId / 银行卡号）
    'accountName'  => '张三',            // 收款人姓名
    // 'bankName'     => '工商银行',      // 银行卡转账时需传
    'clientIp'     => '192.168.1.100',
    'transferDesc' => '用户提现',
    'notifyUrl'    => 'https://your-domain.com/notify/transfer',
    'version'      => '1.0',
    'signType'     => 'MD5',
    'reqTime'      => date('YmdHis'),
];

$params['sign'] = md5Sign($params, $apiKey);

// ========== 发送请求 ==========
$ch = curl_init();
curl_setopt_array($ch, [
    CURLOPT_URL            => $apiBase . '/api/transferOrder',
    CURLOPT_POST           => true,
    CURLOPT_POSTFIELDS     => json_encode($params),
    CURLOPT_HTTPHEADER     => [
        'Content-Type: application/json',
        'Accept: application/json',
    ],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_SSL_VERIFYPEER => false,
    CURLOPT_TIMEOUT        => 30,
]);

$response = curl_exec($ch);
curl_close($ch);

$result = json_decode($response, true);

if (($result['code'] ?? -1) === 0) {
    $data = $result['data'];
    echo "转账发起成功\n";
    echo "transferId: " . $data['transferId'] . "\n";
    echo "mchOrderNo: " . $data['mchOrderNo'] . "\n";
    echo "state: " . $data['state'] . "\n";  // 0-转账生成 1-转账中 2-转账成功 3-转账失败 4-转账退回
} else {
    echo "转账发起失败: " . ($result['msg'] ?? '未知错误') . "\n";
}
```
