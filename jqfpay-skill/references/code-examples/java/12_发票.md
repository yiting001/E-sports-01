# 发票

API路径：

- 开票：`api/invoice/open`
- 查询发票：`api/invoice/query`
- 电子发票URL：`api/invoice/elecUrl`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/74

## 1. 开票

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.invoice.InvoiceOrderCreateReqModel;
import com.jeequan.jeepay.model.invoice.InvoiceOrderCreateResModel;
import com.jeequan.jeepay.request.invoice.InvoiceOrderCreateRequest;
import com.jeequan.jeepay.response.invoice.InvoiceOrderCreateResponse;

public class InvoiceOpenExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        InvoiceOrderCreateReqModel model = new InvoiceOrderCreateReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setMchOrderNo("FP" + System.currentTimeMillis()); // 商户开票单号（商户侧唯一）
        model.setInvoiceWay((byte) 2);              // 开票方式: 2-获取开票链接
        model.setAmount(100L);                      // 含税开票金额，单位：分
        model.setClientIp("127.0.0.1");             // 客户端IP地址
        model.setNotifyUrl("https://your-domain.com/notify/invoice"); // 异步通知地址

        // 购方信息（JSON字符串）
        model.setBuyerInfo("{\"buyerName\":\"测试公司\",\"taxNo\":\"91110108MA01XXXXXX\",\"pushPhone\":\"13800138000\"}");

        // 可选参数
        // model.setExpiredDay(7);                   // 开票链接有效期（天）
        // model.setSellerInfo("{}");                // 销方信息
        // model.setGoodsList("[]");                 // 商品信息
        // model.setChannelExtra("");                // 特定渠道额外参数
        // model.setRemark("备注信息");              // 备注

        // 2. 构建请求
        InvoiceOrderCreateRequest request = new InvoiceOrderCreateRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            InvoiceOrderCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("开票失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            InvoiceOrderCreateResModel resModel = response.get();
            System.out.println("发票单号: " + resModel.getInvoiceOrderId());
            System.out.println("开票状态: " + resModel.getState());  // 0-订单生成 1-开票中 2-开票成功 3-开票失败 4-开票任务关闭 5-已冲红 6-已作废
            System.out.println("开票链接: " + resModel.getChannelInvoiceQrcodeUrl());
            System.out.println("链接失效时间: " + resModel.getExpiredTime());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

## 2. 查询发票

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.invoice.InvoiceOrderQueryReqModel;
import com.jeequan.jeepay.model.invoice.InvoiceOrderQueryResModel;
import com.jeequan.jeepay.request.invoice.InvoiceOrderQueryRequest;
import com.jeequan.jeepay.response.invoice.InvoiceOrderQueryResponse;

public class InvoiceQueryExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        InvoiceOrderQueryReqModel model = new InvoiceOrderQueryReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setInvoiceOrderId("FP2026052600001001"); // 发票单号（与mchOrderNo二选一）
        // model.setMchOrderNo("FP1716700800000");   // 商户开票单号

        // 2. 构建请求
        InvoiceOrderQueryRequest request = new InvoiceOrderQueryRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            InvoiceOrderQueryResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("查询失败: " + response.getMsg());
                return;
            }

            InvoiceOrderQueryResModel resModel = response.get();
            System.out.println("发票单号: " + resModel.getInvoiceOrderId());
            System.out.println("商户单号: " + resModel.getMchOrderNo());
            System.out.println("开票金额: " + resModel.getAmount() + " 分");
            System.out.println("税额: " + resModel.getFeeAmount() + " 分");
            System.out.println("开票状态: " + resModel.getState());  // 0-订单生成 1-开票中 2-开票成功 3-开票失败 4-开票任务关闭 5-已冲红 6-已作废
            System.out.println("渠道发票单号: " + resModel.getChannelInvoiceOrderNo());
            System.out.println("购方信息: " + resModel.getBuyerInfo());
            System.out.println("发票信息: " + resModel.getInvoiceInfo());
            System.out.println("备注: " + resModel.getRemark());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 开票状态说明

| state | 含义 |
|-------|------|
| 0 | 订单生成 |
| 1 | 开票中 |
| 2 | 开票成功 |
| 3 | 开票失败 |
| 4 | 开票任务关闭 |
| 5 | 已冲红 |
| 6 | 已作废 |
