# 提现

API路径：

- 提现余额查询：`api/cashout/balance/query`
- 创建提现订单：`api/cashout/order/create`
- 查询提现订单：`api/cashout/order/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/77

## 1. 提现余额查询

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.cashout.CashoutBalanceQueryReqModel;
import com.jeequan.jeepay.model.cashout.CashoutBalanceQueryResModel;
import com.jeequan.jeepay.request.cashout.CashoutBalanceQueryRequest;
import com.jeequan.jeepay.response.cashout.CashoutBalanceQueryResponse;

public class CashoutBalanceQueryExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        CashoutBalanceQueryReqModel model = new CashoutBalanceQueryReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setIfCode("dgpay");                   // 支付接口代码

        // 2. 构建请求
        CashoutBalanceQueryRequest request = new CashoutBalanceQueryRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            CashoutBalanceQueryResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("查询失败: " + response.getMsg());
                return;
            }

            CashoutBalanceQueryResModel resModel = response.get();
            System.out.println("当前余额: " + resModel.getBalance() + " 分");
            System.out.println("冻结余额: " + resModel.getFrozenBalance() + " 分");
            System.out.println("总余额: " + resModel.getTotalBalance() + " 分");
            System.out.println("渠道账户余额详情: " + resModel.getChannelBalanceDetail());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

## 2. 创建提现订单

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.cashout.CashoutOrderCreateReqModel;
import com.jeequan.jeepay.model.cashout.CashoutOrderCreateResModel;
import com.jeequan.jeepay.request.cashout.CashoutOrderCreateRequest;
import com.jeequan.jeepay.response.cashout.CashoutOrderCreateResponse;

public class CashoutOrderCreateExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        CashoutOrderCreateReqModel model = new CashoutOrderCreateReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setMchOrderNo("CO" + System.currentTimeMillis()); // 商户提现单号（商户侧唯一）
        model.setAmount(100L);                      // 提现金额，单位：分
        model.setIfCode("dgpay");                   // 支付接口代码
        model.setNotifyUrl("https://your-domain.com/notify/cashout"); // 异步通知地址

        // 可选参数
        // model.setChannelExtra("");                // 特定渠道额外参数
        // model.setRemark("测试提现备注");           // 备注

        // 2. 构建请求
        CashoutOrderCreateRequest request = new CashoutOrderCreateRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            CashoutOrderCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("提现失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            CashoutOrderCreateResModel resModel = response.get();
            System.out.println("提现订单号: " + resModel.getCashoutOrderId());
            System.out.println("商户单号: " + resModel.getMchOrderNo());
            System.out.println("提现状态: " + resModel.getState());  // 0-提现单创建 1-提现中 2-提现成功 3-提现失败

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 创建提现订单成功响应

```json
{
    "code": 0,
    "msg": "SUCCESS",
    "sign": "A1B2C3D4E5F6...",

        "mchOrderNo": "CO1716700800000",
        "state": 0,
```

## 3. 查询提现订单

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.cashout.CashoutOrderQueryReqModel;
import com.jeequan.jeepay.model.cashout.CashoutOrderQueryResModel;
import com.jeequan.jeepay.request.cashout.CashoutOrderQueryRequest;
import com.jeequan.jeepay.response.cashout.CashoutOrderQueryResponse;

public class CashoutOrderQueryExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        CashoutOrderQueryReqModel model = new CashoutOrderQueryReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setMchOrderNo("CO1716700800000");     // 商户提现单号（与cashoutOrderId二选一）
        // model.setCashoutOrderId("C2026052600001001"); // 支付系统提现订单号

        // 2. 构建请求
        CashoutOrderQueryRequest request = new CashoutOrderQueryRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            CashoutOrderQueryResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("查询失败: " + response.getMsg());
                return;
            }

            CashoutOrderQueryResModel resModel = response.get();
            System.out.println("提现订单号: " + resModel.getCashoutOrderId());
            System.out.println("商户单号: " + resModel.getMchOrderNo());
            System.out.println("支付接口代码: " + resModel.getIfCode());
            System.out.println("提现金额: " + resModel.getAmount() + " 分");
            System.out.println("提现前余额: " + resModel.getBalance() + " 分");
            System.out.println("提现状态: " + resModel.getState());  // 0-提现单创建 1-提现中 2-提现成功 3-提现失败
            System.out.println("渠道提现单号: " + resModel.getChannelOrderNo());
            System.out.println("提现成功时间: " + resModel.getSuccessTime());
            System.out.println("创建时间: " + resModel.getCreatedAt());
            System.out.println("提现银行: " + resModel.getBankName());
            System.out.println("提现账户: " + resModel.getAccountNo());
            System.out.println("账户名称: " + resModel.getAccountName());
            System.out.println("备注: " + resModel.getRemark());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
