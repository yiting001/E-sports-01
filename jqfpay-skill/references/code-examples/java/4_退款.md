# 退款

API路径：`api/refund/refundOrder`

## SDK调用示例（Java）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.RefundOrderCreateReqModel;
import com.jeequan.jeepay.model.RefundOrderCreateResModel;
import com.jeequan.jeepay.request.RefundOrderCreateRequest;
import com.jeequan.jeepay.response.RefundOrderCreateResponse;

public class RefundOrderCreateExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        RefundOrderCreateReqModel model = new RefundOrderCreateReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setPayOrderId("P2026052100001234");   // 原支付订单号
        model.setMchRefundNo("R" + System.currentTimeMillis()); // 商户退款单号（商户侧唯一）
        model.setRefundAmount(50L);                 // 退款金额，单位：分（支持部分退款）
        model.setRefundReason("用户申请退款");       // 退款原因
        model.setCurrency("CNY");                   // 货币代码

        // 通知地址（可选，不传时使用应用默认配置）
        model.setNotifyUrl("https://your-domain.com/notify/refund");

        // 2. 构建请求
        RefundOrderCreateRequest request = new RefundOrderCreateRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            RefundOrderCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("退款申请失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            RefundOrderCreateResModel resModel = response.get();
            System.out.println("退款订单号: " + resModel.getRefundOrderId());
            System.out.println("退款状态: " + resModel.getState());       // 0-退款生成 1-退款中 2-退款成功 3-退款失败
            System.out.println("渠道退款单号: " + resModel.getChannelOrderNo());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
