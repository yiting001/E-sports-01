# 关闭订单

API路径：`api/pay/close`

## SDK调用示例（Java）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.PayOrderCloseReqModel;
import com.jeequan.jeepay.model.PayOrderCloseResModel;
import com.jeequan.jeepay.request.PayOrderCloseRequest;
import com.jeequan.jeepay.response.PayOrderCloseResponse;

public class PayOrderCloseExample {

    public static void main(String[] args) {

        // 1. 构建请求模型（mchOrderNo 与 payOrderId 二选一）
        PayOrderCloseReqModel model = new PayOrderCloseReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setPayOrderId("P2026052100001234");   // 计全付订单号（二选一）
        // model.setMchOrderNo("M1716249600000");   // 商户订单号（二选一）

        // 2. 构建请求
        PayOrderCloseRequest request = new PayOrderCloseRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            PayOrderCloseResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("关闭失败: " + response.getMsg());
                return;
            }

            // 4. 验签
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            PayOrderCloseResModel resModel = response.get();
            System.out.println("关闭结果: " + resModel.getState());
            System.out.println("渠道错误码: " + resModel.getChannelErrCode());
            System.out.println("渠道错误信息: " + resModel.getChannelErrMsg());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
