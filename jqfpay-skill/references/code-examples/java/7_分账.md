# 分账

分账包含两个主要操作：**绑定分账接收方** 和 **发起分账执行**。

API路径：
- 绑定接收方：`api/divisionReceiver/bind`
- 执行分账：`api/division/exec`

## SDK调用示例（Java）

### 1. 绑定分账接收方

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.DivisionReceiverBindReqModel;
import com.jeequan.jeepay.model.DivisionReceiverBindResModel;
import com.jeequan.jeepay.request.DivisionReceiverBindRequest;
import com.jeequan.jeepay.response.DivisionReceiverBindResponse;

public class DivisionReceiverBindExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        DivisionReceiverBindReqModel model = new DivisionReceiverBindReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setIfCode("WXPAY");                   // 支付接口代码
        model.setAccType(1);                        // 账户类型：1-个人 2-企业
        model.setAccNo("receiver_openid_xxx");      // 接收方账号（微信openid / 支付宝登录号）
        model.setAccName("张三");                    // 接收方姓名
        model.setDivisionProfit("0.30");            // 分账比例（百分比字符串，如"30.00"表示30%）
        model.setRelationType("PARTNER");           // 关系类型：SERVICE_PROVIDER / PARTNER / STORE 等

        // 结算账户信息（可选）
        // model.setSettAccountNo("6222021234567890123");
        // model.setSettAccountName("张三");
        // model.setSettAccountBankName("中国工商银行");
        // model.setSettAccountBankBranchName("北京朝阳支行");
        // model.setSettAccountBankBranchCode("102100000000");

        // 2. 构建请求
        DivisionReceiverBindRequest request = new DivisionReceiverBindRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            DivisionReceiverBindResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("绑定失败: " + response.getMsg());
                return;
            }

            DivisionReceiverBindResModel resModel = response.get();
            System.out.println("接收方ID: " + resModel.getReceiverId());
            System.out.println("接收方别名: " + resModel.getReceiverAlias());
            System.out.println("绑定状态: " + resModel.getState());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 2. 发起分账执行

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.PayOrderDivisionExecReqModel;
import com.jeequan.jeepay.model.PayOrderDivisionExecResModel;
import com.jeequan.jeepay.request.PayOrderDivisionExecRequest;
import com.jeequan.jeepay.response.PayOrderDivisionExecResponse;

public class PayOrderDivisionExecExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        PayOrderDivisionExecReqModel model = new PayOrderDivisionExecReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setPayOrderId("P2026052100001234");   // 原支付订单号
        // model.setMchOrderNo("M1716249600000");   // 或使用商户订单号
        // model.setAutoFinish(true);               // 是否自动完结分账

        // 2. 构建请求
        PayOrderDivisionExecRequest request = new PayOrderDivisionExecRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            PayOrderDivisionExecResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("分账失败: " + response.getMsg());
                return;
            }

            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            PayOrderDivisionExecResModel resModel = response.get();
            System.out.println("分账批次号: " + resModel.getBatchOrderId());
            System.out.println("分账状态: " + resModel.getState());  // 1-待分账 2-分账中 3-分账成功 4-分账失败
            System.out.println("渠道批次号: " + resModel.getChannelBatchOrderId());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
