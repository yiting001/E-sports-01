# 统一支付下单

API路径：`api/pay/unifiedOrder`

## SDK调用示例（Java）

```java
import com.alibaba.fastjson.JSONObject;
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.PayOrderCreateReqModel;
import com.jeequan.jeepay.model.PayOrderCreateResModel;
import com.jeequan.jeepay.request.PayOrderCreateRequest;
import com.jeequan.jeepay.response.PayOrderCreateResponse;

public class PayOrderCreateExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        PayOrderCreateReqModel model = new PayOrderCreateReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setMchOrderNo("M" + System.currentTimeMillis()); // 商户订单号（商户侧唯一）
        model.setWayCode("WX_BAR");                 // 支付方式：微信条码支付
        model.setAmount(100L);                      // 金额，单位：分
        model.setCurrency("CNY");                   // 货币代码
        model.setClientIp("192.168.1.100");         // 客户端IP
        model.setSubject("测试商品标题");             // 商品标题
        model.setBody("测试商品描述");               // 商品描述
        model.setNotifyUrl("https://your-domain.com/notify/pay"); // 异步通知地址
        model.setReturnUrl("https://your-domain.com/return");     // 前端跳转地址
        model.setDivisionMode((byte) 0);            // 分账模式：0-不允许分账

        // 2. 设置渠道扩展参数（channelExtra）
        //    不同支付方式需要传入不同的扩展参数
        JSONObject channelExtra = new JSONObject();

        // WX_BAR / ALI_BAR 条码支付需要传入 authCode（付款码）
        channelExtra.put("authCode", "134567890123456789");

        // WX_JSAPI / ALI_JSAPI 需要传入 channelUserId（openid / 用户ID）
        // channelExtra.put("channelUserId", "oUpF8uMuAJO_M2pxb1Q9zNjWeS6o");

        // QR_CASHIER / WX_H5 / ALI_WAP 等需要传入 payDataType
        // channelExtra.put("payDataType", "codeImgUrl"); // codeImgUrl / payUrl

        model.setChannelExtra(channelExtra.toJSONString());

        // 3. 构建请求
        PayOrderCreateRequest request = new PayOrderCreateRequest();
        request.setBizModel(model);

        // 4. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";  // 计全付服务地址
        String apiKey  = "yourAppSecret";           // 应用密钥（appSecret）
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            // MD5签名方式
            PayOrderCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("下单失败: code=" + response.getCode() + ", msg=" + response.getMsg());
                return;
            }

            // 5. 获取响应模型
            PayOrderCreateResModel resModel = response.get();
            System.out.println("支付订单号: " + resModel.getPayOrderId());
            System.out.println("支付数据: " + resModel.getPayData());

            // 根据 payDataType 判断返回内容类型：
            // - codeImgUrl: 二维码图片URL，可直接展示
            // - payUrl: 支付跳转URL，重定向即可
            // - codeUrl: 二维码内容字符串，需自行生成二维码

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

    "msg": "SUCCESS",

    "data": {

        "payDataType": "codeUrl",
