# 协议支付（银行协议支付）

协议支付分为三个步骤：**预绑卡** -> **确认绑卡** -> **协议支付扣款**。

API路径：
- 预绑卡：`api/bankAgree/preBind`
- 确认绑卡：`api/bankAgree/confirmBind`
- 协议支付扣款（复用统一下单接口）：`api/pay/unifiedOrder`（wayCode = `BANK_AGREE`）

## SDK调用示例（Java）

### 1. 预绑卡（发送短信验证码）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.BankAgreePreBindReqModel;
import com.jeequan.jeepay.model.BankAgreePreBindResModel;
import com.jeequan.jeepay.request.BankAgreePreBindRequest;
import com.jeequan.jeepay.response.BankAgreePreBindResponse;

public class BankAgreePreBindExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        BankAgreePreBindReqModel model = new BankAgreePreBindReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setIfCode("TLBANK");                  // 支付接口代码（银行通道）
        model.setBankCardNo("6222021234567890123"); // 银行卡号
        model.setBankCardName("张三");              // 持卡人姓名
        model.setCertNo("110101199001011234");      // 身份证号
        model.setPhoneNo("13800138000");            // 手机号
        model.setUserId("user_001");                // 商户侧用户ID

        // 2. 构建请求
        BankAgreePreBindRequest request = new BankAgreePreBindRequest();
        request.setBizModel(model);

        // 3. 发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            BankAgreePreBindResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("预绑卡失败: " + response.getMsg());
                return;
            }

            BankAgreePreBindResModel resModel = response.get();
            System.out.println("绑卡流水号: " + resModel.getBindId());
            System.out.println("协议编号: " + resModel.getAgreeRecordId());
            // 用户此时会收到银行短信验证码，用于下一步确认绑卡

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 2. 确认绑卡（验证短信验证码）

```java
import com.jeequan.jeepay.JeepayClient;

        BankAgreeConfirmBindReqModel model = new BankAgreeConfirmBindReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setIfCode("TLBANK");
        model.setAgreeRecordId("AGR202605210001");  // 预绑卡返回的协议编号
        model.setSmsCode("123456");                 // 用户收到的短信验证码

        // 2. 构建请求
        BankAgreeConfirmBindRequest request = new BankAgreeConfirmBindRequest();
        request.setBizModel(model);

        // 3. 发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            BankAgreeConfirmBindResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("确认绑卡失败: " + response.getMsg());
                return;
            }

            BankAgreeConfirmBindResModel resModel = response.get();
            System.out.println("绑卡状态: " + resModel.getState());
            System.out.println("协议编号: " + resModel.getAgreeRecordId());
            System.out.println("银行卡掩码: " + resModel.getBankCardNo());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 3. 协议支付扣款（复用统一下单接口）

```java
import com.alibaba.fastjson.JSONObject;
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.PayOrderCreateReqModel;
import com.jeequan.jeepay.model.PayOrderCreateResModel;
import com.jeequan.jeepay.request.PayOrderCreateRequest;
import com.jeequan.jeepay.response.PayOrderCreateResponse;

public class BankAgreeProtocolpayExample {

    public static void main(String[] args) {

        // 1. 构建请求模型 - 使用统一下单接口，wayCode 为 BANK_AGREE
        PayOrderCreateReqModel model = new PayOrderCreateReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setMchOrderNo("M" + System.currentTimeMillis());
        model.setWayCode("BANK_AGREE");             // 协议支付方式
        model.setAmount(100L);
        model.setCurrency("CNY");
        model.setClientIp("192.168.1.100");
        model.setSubject("协议支付测试");
        model.setBody("协议支付测试");
        model.setNotifyUrl("https://your-domain.com/notify/pay");

        // 2. 在 channelExtra 中传入协议编号
        JSONObject channelExtra = new JSONObject();
        channelExtra.put("agreeRecordId", "AGR202605210001");  // 绑卡成功后返回的协议编号
        model.setChannelExtra(channelExtra.toJSONString());

        // 3. 构建请求并发起
        PayOrderCreateRequest request = new PayOrderCreateRequest();
        request.setBizModel(model);

        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            PayOrderCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("协议支付失败: " + response.getMsg());
                return;
            }

            PayOrderCreateResModel resModel = response.get();
            System.out.println("支付订单号: " + resModel.getPayOrderId());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
