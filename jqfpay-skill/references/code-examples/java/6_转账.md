# 转账

API路径：`api/transferOrder`

## SDK调用示例（Java）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.TransferOrderCreateReqModel;
import com.jeequan.jeepay.model.TransferOrderCreateResModel;
import com.jeequan.jeepay.request.TransferOrderCreateRequest;
import com.jeequan.jeepay.response.TransferOrderCreateResponse;

public class TransferOrderCreateExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        TransferOrderCreateReqModel model = new TransferOrderCreateReqModel();
        model.setMchNo("M1679428700001");           // 商户号
        model.setAppId("66372b9c09992a00081e52ae"); // 应用ID
        model.setMchOrderNo("T" + System.currentTimeMillis()); // 商户转账单号（商户侧唯一）
        model.setIfCode("WXPAY");                   // 支付接口代码（如：WXPAY / ALIPAY 等）
        model.setEntryType("WX_CASH");              // 入账类型：WX_CASH-微信零钱 / ALI_CASH-支付宝余额 / BANK_CARD-银行卡
        model.setAmount(10000L);                    // 转账金额，单位：分
        model.setCurrency("CNY");                   // 货币代码
        model.setAccountNo(" receiver_openid_xxx"); // 收款账号（微信openid / 支付宝账号 / 银行卡号）
        model.setAccountName("张三");               // 收款人姓名
        model.setTransferDesc("用户提现");          // 转账说明
        model.setNotifyUrl("https://your-domain.com/notify/transfer"); // 异步通知地址

        // 银行卡转账时需额外设置银行信息
        // model.setBankName("中国工商银行");
        // model.setBankBranchName("北京朝阳支行");
        // model.setBankCode("ICBC");

        // 2. 构建请求
        TransferOrderCreateRequest request = new TransferOrderCreateRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            TransferOrderCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("转账失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            TransferOrderCreateResModel resModel = response.get();
            System.out.println("转账订单号: " + resModel.getTransferId());
            System.out.println("转账状态: " + resModel.getState());      // 0-订单生成 1-转账中 2-转账成功 3-转账失败 4-订单关闭
            System.out.println("渠道订单号: " + resModel.getChannelOrderNo());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
