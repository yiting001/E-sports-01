# 查询订单

API路径：`api/pay/query`

## SDK调用示例（Java）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.PayOrderQueryReqModel;
import com.jeequan.jeepay.model.PayOrderQueryResModel;
import com.jeequan.jeepay.request.PayOrderQueryRequest;
import com.jeequan.jeepay.response.PayOrderQueryResponse;

public class PayOrderQueryExample {

    public static void main(String[] args) {

        // 1. 构建请求模型（mchOrderNo 与 payOrderId 二选一）
        PayOrderQueryReqModel model = new PayOrderQueryReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setPayOrderId("P2026052100001234");   // 计全付订单号（二选一）
        // model.setMchOrderNo("M1716249600000");   // 商户订单号（二选一）

        // 2. 构建请求
        PayOrderQueryRequest request = new PayOrderQueryRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            PayOrderQueryResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("查询失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应模型
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            PayOrderQueryResModel resModel = response.get();
            System.out.println("订单号: " + resModel.getPayOrderId());
            System.out.println("订单状态: " + resModel.getState());       // 0-订单生成 1-支付中 2-支付成功 3-支付失败 4-已撤销 5-已退款 6-订单关闭
            System.out.println("支付金额: " + resModel.getAmount());      // 单位：分
            System.out.println("支付方式: " + resModel.getWayCode());
            System.out.println("渠道订单号: " + resModel.getChannelOrderNo());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
