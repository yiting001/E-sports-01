# 钱包

API路径：

| 接口 | API路径 |
|------|---------|
| 创建钱包 | `api/wallet/user` |
| 绑定银行卡 | `api/wallet/bankcard/bind` |
| 查询钱包用户 | `api/wallet/user/query` |
| 钱包变动明细 | `api/wallet/user/changeDetails` |
| 钱包提现 | `api/wallet/cashout` |
| 查询钱包提现 | `api/wallet/cashout/query` |
| 钱包转账 | `api/wallet/transfer` |
| 查询钱包转账 | `api/wallet/transfer/query` |

## 1. 创建钱包

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.WalletCreateReqModel;
import com.jeequan.jeepay.model.WalletCreateResModel;
import com.jeequan.jeepay.request.WalletCreateRequest;
import com.jeequan.jeepay.response.WalletCreateResponse;

public class WalletCreateExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        WalletCreateReqModel model = new WalletCreateReqModel();
        model.setMchNo("M1679428700001");                    // 商户号
        model.setAppId("66372b9c09992a00081e52ae");          // 应用ID
        model.setMchOrderNo("W" + System.currentTimeMillis()); // 商户订单号
        model.setIfCode("hnawalletpay");                     // 支付接口代码，目前仅支持：hnawalletpay
        model.setIdcardNo("412928194606080028");              // 身份证号
        model.setIdcardName("张三");                          // 身份证姓名
        model.setTelphone("15254587488");                     // 手机号
        model.setAccountNo("6217000045734385345");            // 银行卡号
        model.setBankName("中国建设银行");                     // 银行名称
        model.setBankMobile("15254587488");                   // 银行预留手机号（可选）

        // 2. 构建请求
        WalletCreateRequest request = new WalletCreateRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            WalletCreateResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("创建钱包失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            WalletCreateResModel resModel = response.get();
            System.out.println("钱包用户编号: " + resModel.getWalletId());
            System.out.println("开户状态: " + resModel.getChannelState());
            // channelState: 0-未开户 1-开户成功 2-待审核 3-审核拒绝 4-待激活 5-开户失败 6-已注销

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

        model.setMchNo("M1679428700001");                    // 商户号
        model.setAppId("66372b9c09992a00081e52ae");          // 应用ID
        model.setTransferId("T20260526000001");               // 平台转账单号（与mchOrderNo二选一）
        // model.setMchOrderNo("WT1716249600000");            // 商户订单号

        // 2. 构建请求
        WalletTransferQueryRequest request = new WalletTransferQueryRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            WalletTransferQueryResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("查询转账失败: " + response.getMsg());
                return;
            }

            WalletTransferQueryResModel resModel = response.get();
            System.out.println("转账单号: " + resModel.getTransferId());
            System.out.println("商户订单号: " + resModel.getMchOrderNo());
            System.out.println("付款方钱包: " + resModel.getPayWalletId());
            System.out.println("收款方钱包: " + resModel.getReceiveWalletId());
            System.out.println("转账金额: " + resModel.getAmount() + " 分");
            System.out.println("转账状态: " + resModel.getState());
            // state: 0-订单生成 1-转账中 2-转账成功 3-转账失败 4-订单关闭
            System.out.println("渠道转账单号: " + resModel.getChannelOrderNo());
            System.out.println("转账成功时间: " + resModel.getSuccessTime());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
