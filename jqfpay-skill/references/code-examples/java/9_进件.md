# 进件（商户入驻）

进件流程包含多个步骤：**上传文件** -> **发起进件** -> **查询进件状态** -> **签约详情** -> **实名查询** -> **微信配置**。

API路径：
- 上传文件：`api/mchApplyment/uploadFile`
- 发起进件：`api/mchApplyment/income`
- 查询进件：`api/mchApplyment/income/query`
- 签约详情：`api/mchApplyment/signInfo`
- 实名查询：`api/mchApplyment/realName/query`
- 微信配置：`api/mchApplyment/wxConfig`

## SDK调用示例（Java）

### 1. 上传文件

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.applyment.ApplymentMchUploadImgReqModel;
import com.jeequan.jeepay.model.applyment.ApplymentMchUploadImgResModel;
import com.jeequan.jeepay.request.applyment.ApplymentMchUploadImgRequest;
import com.jeequan.jeepay.response.applyment.ApplymentMchUploadImgResponse;

public class ApplymentUploadExample {

    public static void main(String[] args) {

        // 1. 构建请求模型
        ApplymentMchUploadImgReqModel model = new ApplymentMchUploadImgReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setFileType("idCardFront");           // 文件类型：idCardFront / idCardBack / license 等
        model.setFileUrl("https://your-oss.com/img/id_front.jpg"); // 图片URL
        // 或者传入 base64
        // model.setFileBase64("data:image/jpeg;base64,/9j/4AAQ...");

        // 2. 构建请求
        ApplymentMchUploadImgRequest request = new ApplymentMchUploadImgRequest();
        request.setBizModel(model);

        // 3. 发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            ApplymentMchUploadImgResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("上传失败: " + response.getMsg());
                return;
            }

            ApplymentMchUploadImgResModel resModel = response.get();
            System.out.println("文件ID: " + resModel.getFileId());
            // 文件ID用于后续进件申请中引用该文件

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 2. 发起进件

```java
import com.alibaba.fastjson.JSONObject;
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.applyment.ApplymentMchInComeReqModel;
import com.jeequan.jeepay.model.applyment.ApplymentMchInComeResModel;
import com.jeequan.jeepay.model.applyment.applymentModel.ApplymentInfo;
import com.jeequan.jeepay.model.applyment.applymentModel.common.IdcardInfo;
import com.jeequan.jeepay.model.applyment.applymentModel.common.MchInfo;
import com.jeequan.jeepay.model.applyment.applymentModel.common.PaywayFee;
import com.jeequan.jeepay.request.applyment.ApplymentMchInComeRequest;
import com.jeequan.jeepay.response.applyment.ApplymentMchInComeResponse;

import java.util.ArrayList;
import java.util.List;

public class ApplymentIncomeExample {

    public static void main(String[] args) {

        // 1. 构建进件信息
        ApplymentInfo applymentInfo = new ApplymentInfo();
        applymentInfo.setMerchantType(1);            // 商户类型：1-个体 2-企业

        // 商户基本信息
        MchInfo mchInfoModel = new MchInfo();
        mchInfoModel.setMchFullName("某某餐饮店");
        mchInfoModel.setMchShortName("餐饮店");
        mchInfoModel.setContactName("张三");
        mchInfoModel.setContactPhone("13800138000");
        mchInfoModel.setContactEmail("zhangsan@example.com");
        mchInfoModel.setAddress("北京市朝阳区某某路100号");
        applymentInfo.setMchInfo(mchInfoModel);

        // 身份证信息（引用上传文件后返回的fileId）
        IdcardInfo idcardInfo = new IdcardInfo();
        idcardInfo.setIdType("ID_CARD");
        idcardInfo.setIdCardNo("110101199001011234");
        idcardInfo.setIdCardName("张三");
        idcardInfo.setIdCardFrontImgId("file_uuid_front_001");
        idcardInfo.setIdCardBackImgId("file_uuid_back_001");
        idcardInfo.setIdCardEffectiveDate("2020-01-01");
        idcardInfo.setIdCardExpireDate("2040-01-01");
        applymentInfo.setIdcardInfo(idcardInfo);

        // 费率配置
        List<PaywayFee> paywayFeeList = new ArrayList<>();
        PaywayFee wxFee = new PaywayFee();
        wxFee.setWayCode("WX_JSAPI");
        wxFee.setRate("0.60");                      // 费率：0.60%
        paywayFeeList.add(wxFee);
        applymentInfo.setPaywayFeeList(paywayFeeList);

        // 2. 构建请求模型
        ApplymentMchInComeReqModel model = new ApplymentMchInComeReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setIfCode("WXPAY");                   // 入驻的支付渠道
        model.setApplymentInfo(JSONObject.toJSONString(applymentInfo));

        // 3. 构建请求
        ApplymentMchInComeRequest request = new ApplymentMchInComeRequest();
        request.setBizModel(model);

        // 4. 发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            ApplymentMchInComeResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("进件失败: " + response.getMsg());
                return;
            }

            ApplymentMchInComeResModel resModel = response.get();
            System.out.println("申请单号: " + resModel.getApplyId());
            System.out.println("审核状态: " + resModel.getState());
            // 状态: 0-保存 1-审核中 2-审核通过 3-审核驳回 4-待商户验证 5-待商户签约

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```

### 6. 微信配置（wxConfig）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.applyment.ApplymentMchWxConfigReqModel;
import com.jeequan.jeepay.model.applyment.ApplymentMchWxConfigResModel;
import com.jeequan.jeepay.request.applyment.ApplymentMchWxConfigRequest;
import com.jeequan.jeepay.response.applyment.ApplymentMchWxConfigResponse;

public class ApplymentWxConfigExample {

    public static void main(String[] args) {

        ApplymentMchWxConfigReqModel model = new ApplymentMchWxConfigReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setApplyId("AP2026052100001");
        model.setWxAppId("wx1234567890abcdef");     // 微信公众号/小程序的AppID
        model.setWxAppType("JSAPI");                // 类型：JSAPI / MINIAPP

        ApplymentMchWxConfigRequest request = new ApplymentMchWxConfigRequest();
        request.setBizModel(model);

        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            ApplymentMchWxConfigResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("配置失败: " + response.getMsg());
                return;
            }

            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            ApplymentMchWxConfigResModel resModel = response.get();
            System.out.println("配置状态: " + resModel.getState());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
