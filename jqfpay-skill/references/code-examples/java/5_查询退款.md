# 查询退款

API路径：`api/refund/query`

## SDK调用示例（Java）

```java
import com.jeequan.jeepay.JeepayClient;
import com.jeequan.jeepay.exception.JeepayException;
import com.jeequan.jeepay.model.RefundOrderQueryReqModel;
import com.jeequan.jeepay.model.RefundOrderQueryResModel;
import com.jeequan.jeepay.request.RefundOrderQueryRequest;
import com.jeequan.jeepay.response.RefundOrderQueryResponse;

public class RefundOrderQueryExample {

    public static void main(String[] args) {

        // 1. 构建请求模型（mchRefundNo 与 refundOrderId 二选一）
        RefundOrderQueryReqModel model = new RefundOrderQueryReqModel();
        model.setMchNo("M1679428700001");
        model.setAppId("66372b9c09992a00081e52ae");
        model.setRefundOrderId("R2026052100005678");  // 计全付退款订单号（二选一）
        // model.setMchRefundNo("R1716249600000");    // 商户退款单号（二选一）

        // 2. 构建请求
        RefundOrderQueryRequest request = new RefundOrderQueryRequest();
        request.setBizModel(model);

        // 3. 创建客户端并发起请求
        String apiBase = "https://pay.jeepay.cn";
        String apiKey  = "yourAppSecret";
        JeepayClient client = new JeepayClient(apiBase, apiKey);

        try {
            RefundOrderQueryResponse response = client.execute(request);

            if (response.getCode() != 0) {
                System.err.println("查询失败: " + response.getMsg());
                return;
            }

            // 4. 验签并获取响应
            if (!response.isSuccess(apiKey)) {
                System.err.println("验签失败");
                return;
            }

            RefundOrderQueryResModel resModel = response.get();
            System.out.println("退款订单号: " + resModel.getRefundOrderId());
            System.out.println("退款状态: " + resModel.getState());          // 0-退款生成 1-退款中 2-退款成功 3-退款失败
            System.out.println("退款金额: " + resModel.getRefundAmount());   // 单位：分
            System.out.println("渠道退款单号: " + resModel.getChannelOrderNo());

        } catch (JeepayException e) {
            System.err.println("请求异常: " + e.getMessage());
        }
    }
}
```
