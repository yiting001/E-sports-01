# 进件

API路径：`api/mchApplyment/income` + `api/mchApplyment/uploadFile` + `api/mchApplyment/income/query`

在线文档：
- 商户进件：https://doc.jeequan.com/#/integrate/jqf/api/343
- 上传文件：https://doc.jeequan.com/#/integrate/jqf/api/344
- 查询进件：https://doc.jeequan.com/#/integrate/jqf/api/347
- 签约信息：https://doc.jeequan.com/#/integrate/jqf/api/353

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
商户进件完整流程示例
流程：上传文件 -> 提交进件 -> 轮询状态 -> 签约 -> 实名认证 -> 微信配置
"""

from jeepay import AppConfig, Income

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")       # 服务商商户号
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 步骤一：上传证件图片（获取 fileId）
# ============================================================
upload_result = Income.income_upload(
    # 文件上传方式（SDK内部处理文件读取和multipart上传）
    fileName="id_card_front.jpg",                # 文件名
    filePath="/path/to/id_card_front.jpg",       # 文件本地路径
    # 或者使用文件内容：
    # fileContent=open("/path/to/id_card_front.jpg", "rb").read(),
    # fileName="id_card_front.jpg",
)
# 注意：实际参数可能因SDK版本不同而有差异，请参考SDK源码

if upload_result.get("code") == 0:
    upload_data = upload_result.get("data", {})
    file_id = upload_data.get("fileId")
    print(f"文件上传成功！fileId: {file_id}")
else:
    print(f"文件上传失败！code={upload_result.get('code')}, msg={upload_result.get('msg')}")

# 上传其他所需文件（身份证反面、营业执照等）
# upload_result_back = Income.income_upload(...)
# upload_result_license = Income.income_upload(...)


# ============================================================
# 步骤二：提交进件申请
# ============================================================
income_result = Income.income_launch(
    # 商户基本信息
    mchInfo='{
        "mchShortName": "测试商户",
        "mchName": "测试商户有限公司",
        "mchType": 1,
        "contactName": "张三",
        "contactPhone": "18600001111",
        "contactEmail": "test@example.com"
    }',

    # 进件详细信息
    applymentInfo='{
        "businessLicenseType": 1,
        "businessLicenseNo": "91110000MA12345678",
        "businessLicenseImg": "' + file_id + '",
        "idType": "ID_CARD",
        "idCardName": "张三",
        "idCardNo": "110101199001011234",
        "idCardFrontImg": "' + file_id + '",
        "idCardBackImg": "' + file_id + '",
        "settleAccountNo": "6222021234567890123",
        "settleAccountName": "张三",
        "settleAccountBank": "工商银行",
        "settleAccountType": 1
    }',

    notifyUrl="https://your-domain.com/api/notify/income",  # 进件结果异步通知地址
)

if income_result.get("code") == 0:
    income_data = income_result.get("data", {})
    applyment_id = income_data.get("applymentId")
    print(f"进件申请成功！applymentId: {applyment_id}")
else:
    print(f"进件申请失败！code={income_result.get('code')}, msg={income_result.get('msg')}")


# ============================================================
# 步骤三：查询进件状态（建议轮询，或通过异步通知获取）
# ============================================================
import time

max_retry = 10
for i in range(max_retry):
    query_result = Income.income_query(
        applymentId=applyment_id,          # 进件申请ID
    )

    if query_result.get("code") == 0:
        query_data = query_result.get("data", {})
        apply_state = query_data.get("applyState")
        print(f"进件状态: {apply_state}")
        # applyState 说明：
        #   0 - 审核中
        #   1 - 审核通过（需签约）
        #   2 - 审核驳回
        #   3 - 已签约（进件完成）
        #   4 - 已拒绝

        if apply_state == 1:
            print("审核通过！可以进行签约。")
            break
        elif apply_state == 3:
            print("进件完成！商户已签约。")
            break
        elif apply_state in (2, 4):
            print(f"进件失败: {query_data.get('applyMsg')}")
            break
    else:
        print(f"查询失败: {query_result.get('msg')}")

    if i < max_retry - 1:
        print("等待5秒后重试...")
        time.sleep(5)


# ============================================================
# 步骤四：获取签约信息（审核通过后）
# ============================================================
sign_result = Income.sign_info_query(
    applymentId=applyment_id,
)
if sign_result.get("code") == 0:
    sign_data = sign_result.get("data", {})
    sign_url = sign_data.get("signUrl")
    print(f"签约URL: {sign_url}")
    print("请引导商户访问该URL完成签约")


# ============================================================
# 步骤五：实名认证查询
# ============================================================
realname_result = Income.real_name_query(
    applymentId=applyment_id,
)
if realname_result.get("code") == 0:
    realname_data = realname_result.get("data", {})
    print(f"实名认证信息: {realname_data}")


# ============================================================
# 步骤六：微信支付参数配置（签约完成后）
# ============================================================
wx_config_result = Income.wx_params_config(
    applymentId=applyment_id,
    # 微信支付配置参数
    subMchId="sub_mch_id_from_sign",       # 签约后获取的子商户号
    # 其他微信配置参数根据实际需求填写
)
if wx_config_result.get("code") == 0:
    print("微信配置成功！")
```

```json
    "code": 0,
    "msg": "SUCCESS",
    "data": {
        "fileId": "FILE202605211900000001",
        "fileName": "id_card_front.jpg"
    }
}
```

```json

    "code": 0,
    "msg": "SUCCESS",
    "data": {
        "applymentId": "AP202605211900000001",

        "msg": "提交成功，等待审核"
    }
}
```

