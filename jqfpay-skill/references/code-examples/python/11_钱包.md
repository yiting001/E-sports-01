# 钱包

API路径：

| 接口 | API路径 |
|------|---------|
| 创建钱包 | `api/wallet/user` |
| 绑定银行卡 | `api/wallet/bankcard/bind` |
| 查询钱包用户 | `api/wallet/user/query` |
| 钱包变动明细 | `api/wallet/user/changeDetails` |
| 钱包提现 | `api/wallet/cashout` |
| 查询钱包提现 | `api/wallet/cashout/query` |
| 钱包转账 | `api/wallet/transfer` |
| 查询钱包转账 | `api/wallet/transfer/query` |

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
钱包接口示例
包含：创建钱包、绑定银行卡、查询钱包用户、钱包变动明细、钱包提现、查询提现、钱包转账、查询转账
"""

import time
import jeepay
from jeepay.app_config import AppConfig

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 创建钱包（钱包开户）
# ============================================================
print("=== 创建钱包 ===")
try:
    # 创建钱包需要使用 v2.0 接口版本
    AppConfig.set_version("2.0")
    result = jeepay.Wallet.user_wallet(
        mchOrderNo=f"mho{round(time.time() * 1000)}",  # 商户订单号
        ifCode="hnawalletpay",         # 支付接口代码，目前仅支持：hnawalletpay
        idcardNo="412928194606080028", # 身份证号
        idcardName="张三",             # 身份证姓名
        telphone="15254587488",        # 手机号
        accountNo="6217000045734385345",  # 银行卡号
        bankName="中国建设银行",        # 银行名称
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 3. 绑定银行卡
# ============================================================
print("\n=== 绑定银行卡 ===")
try:
    AppConfig.set_version("2.0")
    result = jeepay.Wallet.bankcard_bind(
        mchOrderNo=f"mho{round(time.time() * 1000)}",  # 商户订单号
        walletId="2024102414555816",    # 钱包用户编号，通过钱包开户获取
        accountNo="6217000045734385345", # 银行卡号
        accountName="张三",              # 开户名称
        bankName="中国建设银行",          # 银行名称
        bankMobile="15254587488"         # 银行预留手机号
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 4. 查询钱包用户
# ============================================================
print("\n=== 查询钱包用户 ===")
try:
    result = jeepay.Wallet.user_query_wallet(
        walletId="2024102414555816"     # 钱包用户编号，通过钱包开户获取
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 5. 查询钱包变动明细
# ============================================================
print("\n=== 查询钱包变动明细 ===")
try:
    result = jeepay.Wallet.change_details_wallet(
        walletId="2024102414555816",     # 钱包用户编号
        startDate="",                    # 查询开始时间，格式：YYYY-MM-DD（可选，默认近30天）
        endDate="",                      # 查询结束时间，格式：YYYY-MM-DD（可选，默认近30天）
        pageNum=1                        # 请求页数
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 6. 钱包提现
# ============================================================
print("\n=== 钱包提现 ===")
try:
    result = jeepay.Wallet.cashout_wallet(
        mchOrderNo=f"mho{round(time.time() * 1000)}",  # 商户订单号，需保证唯一
        walletId="2023122611374100",     # 提现钱包用户编号
        accountNo="62175738495739574662", # 提现到账银行卡号
        amount=1,                         # 提现金额，单位：分
        currency="cny",                   # 货币代码
        clientIp="127.0.0.1"             # 客户端IP地址
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 7. 查询钱包提现
# ============================================================
print("\n=== 查询钱包提现 ===")
try:
    result = jeepay.Wallet.cashout_query_wallet(
        cashoutId="C1955463383205658626",  # 平台提现单号（与mchOrderNo二选一）
        # mchOrderNo="20160427210604000490" # 商户订单号
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 8. 钱包转账
# ============================================================
print("\n=== 钱包转账 ===")
try:
    result = jeepay.Wallet.transfer_wallet(
        mchOrderNo=f"mho{round(time.time() * 1000)}",  # 商户订单号
        payWalletId="2024102414555816",    # 付款方钱包用户编号
        receiveWalletId="2024102414550715", # 收款方钱包用户编号
        amount=1,                           # 转账金额，单位：分
        currency="cny",                     # 货币代码
        clientIp="127.0.0.1"               # 客户端IP地址
    )
    print(result)
except Exception as e:
    print(e)

# ============================================================
# 9. 查询钱包转账
# ============================================================
print("\n=== 查询钱包转账 ===")
try:
    result = jeepay.Wallet.transfer_query_wallet(
        transferId="T1955464233865035778",  # 平台转账单号（与mchOrderNo二选一）
        # mchOrderNo="mho1755053978132"     # 商户订单号
    )
    print(result)
except Exception as e:
    print(e)
```
