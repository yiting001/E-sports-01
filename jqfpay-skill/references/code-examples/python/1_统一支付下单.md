# 统一支付下单

API路径：`api/pay/unifiedOrder`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/39

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
统一支付下单 - 微信条码支付（WX_BAR）示例

使用前请确保已将 jeepay-sdk-python 目录添加到 Python 路径中。
SDK下载地址：https://gitee.com/jeequan/jeepay-sdk-python
"""

from jeepay import AppConfig, Pay

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")          # 商户号
AppConfig.set_app_id("607cc0a0ee11e11e7c80")  # 应用ID
AppConfig.set_api_key("YourApiKey1234567890")  # API密钥（MD5签名方式）

# 如果使用 RSA2 签名方式，需额外配置私钥：
# AppConfig.set_private("/path/to/your/private_key.pem")

# ============================================================
# 2. 调用统一下单接口 - 微信条码支付
# ============================================================
result = Pay.create(
    mchOrderNo="M" + "20260521143000",  # 商户订单号（需保证唯一）
    wayCode="WX_BAR",                    # 支付方式：微信条码
    amount=100,                           # 金额，单位：分（100 = 1元）
    currency="cny",                       # 币种
    clientIp="192.168.1.100",            # 客户端IP
    subject="测试商品-微信条码支付",         # 商品标题
    body="测试商品描述",                   # 商品描述
    notifyUrl="https://your-domain.com/api/notify/pay",  # 异步通知地址
    returnUrl="https://your-domain.com/order/result",    # 前端跳转地址
    channelExtra='{"authCode":"13920933111042"}',         # 付款码（用户微信付款码）
    # 可选参数：
    # timeExpire="2026-05-21 15:30:00",  # 过期时间
    # divisionMode=0,                     # 分账模式：0-不分账 1-自动分账 2-手动分账
    # divisionReceiverList='[...]',        # 分账接收方列表
)

# ============================================================
# 3. 处理返回结果
# ============================================================
if result.get("code") == 0:
    print("下单成功！")
    order_data = result.get("data", {})
    print(f"支付订单号: {order_data.get('payOrderId')}")
    print(f"商户订单号: {order_data.get('mchOrderNo')}")
    print(f"订单状态: {order_data.get('state')}")  # 0-订单生成 1-支付中 2-支付成功 3-支付失败

    # 条码支付成功时，payDataType 为 "none"，直接完成支付
    pay_data = order_data.get("payData")
    if pay_data:
        print(f"支付数据: {pay_data}")
else:
    print(f"下单失败！code={result.get('code')}, msg={result.get('msg')}")
```
