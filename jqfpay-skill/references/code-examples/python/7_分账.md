# 分账

API路径：`api/division/receiver/bind` + `api/division/exec`

在线文档：
- 绑定分账用户：https://doc.jeequan.com/#/integrate/jqf/api/55
- 发起分账：https://doc.jeequan.com/#/integrate/jqf/api/57

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
分账示例
包含：绑定分账用户 -> 发起分账 -> 查询分账结果 -> 完成分账
完整流程。
"""

from jeepay import AppConfig, Division

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 绑定分账接收方
# ============================================================
bind_result = Division.bind_user(
    recvAppId="wx1234567890abcdef",      # 接收方应用ID（特约商户的公众号/小程序AppId）
    recvMchNo="M1671695487",              # 掓收方商户号
    accType=1,                             # 账户类型：1-商户号
    accNo="M1671695487",                   # 接收方账号（商户号）
    accName="分账商户A",                   # 接收方名称
    relationType="PARTNER",                # 关系类型：SERVICE_PROVIDER/STORE/PARTNER等
    # divisionProfit=0.1,                  # 分账比例（可选，如0.1表示10%）
)

if bind_result.get("code") == 0:
    print("分账接收方绑定成功！")
    print(f"绑定结果: {bind_result.get('data')}")
else:
    print(f"绑定失败！code={bind_result.get('code')}, msg={bind_result.get('msg')}")


# ============================================================
# 3. 查询分账绑定关系
# ============================================================
bind_query_result = Division.bind_query(
    recvAppId="wx1234567890abcdef",
    recvMchNo="M1671695487",
)
if bind_query_result.get("code") == 0:
    print(f"绑定查询结果: {bind_query_result.get('data')}")


# ============================================================
# 4. 发起分账（对已支付成功的订单进行分账）
# ============================================================
exec_result = Division.exec(
    payOrderId="P202605211430000001",     # 支付订单号

    # 分账接收方列表
    receivers='['
    '  {'
    '    "recvAppId": "wx1234567890abcdef",'
    '    "recvMchNo": "M1671695487",'
    '    "accType": 1,'
    '    "accNo": "M1671695487",'
    '    "divisionAmt": 50,'
    '    "divisionDesc": "服务商分账"'
    '  }'
    ']',
    # divisionAmt: 分账金额，单位：分
)

if exec_result.get("code") == 0:
    print("分账发起成功！")
    exec_data = exec_result.get("data", {})
    print(f"分账批次号: {exec_data.get('batchOrderId')}")
    print(f"分账状态: {exec_data.get('state')}")
else:
    print(f"分账发起失败！code={exec_result.get('code')}, msg={exec_result.get('msg')}")


# ============================================================
# 5. 查询分账结果
# ============================================================
query_result = Division.exec_query(
    payOrderId="P202605211430000001",     # 支付订单号
)
if query_result.get("code") == 0:
    query_data = query_result.get("data", {})
    print(f"分账查询结果: {query_data}")


# ============================================================
# 6. 完成分账（结束分账，解除资金冻结）
# ============================================================
finish_result = Division.exec_finish(
    payOrderId="P202605211430000001",     # 支付订单号
)
if finish_result.get("code") == 0:
    print("分账完成！")
    print(f"完成结果: {finish_result.get('data')}")
else:
    print(f"分账完成失败！code={finish_result.get('code')}, msg={finish_result.get('msg')}")
```

### 下单时自动分账

```python
# 在统一支付下单时设置自动分账模式
from jeepay import Pay

result = Pay.create(
    mchOrderNo="M20260521143000",
    wayCode="WX_JSAPI",
    amount=100,
    subject="自动分账测试",
    notifyUrl="https://your-domain.com/api/notify/pay",
    channelExtra='{"openid":"o6BcIwvSiRpfS8e_UyfQNrYuk2LI"}',

    # 分账相关参数
    divisionMode=1,  # 分账模式：0-不分账 1-自动分账 2-手动分账（需后续调用分账接口）

    # 自动分账时的接收方列表（divisionMode=1 时使用）
    divisionReceiverList='['
    '  {'
    '    "recvAppId": "wx1234567890abcdef",'
    '    "recvMchNo": "M1671695487",'
    '    "accType": 1,'
    '    "accNo": "M1671695487",'
    '    "divisionAmt": 50'
    '  }'
    ']',
)
```
