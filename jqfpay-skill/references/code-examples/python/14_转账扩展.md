# 转账扩展

转账扩展接口包含：转账余额查询、转账凭证申请、转账凭证查询、资金转账支付（充值）。

API路径：
- 转账余额查询：`api/transfer/balance/query`
- 转账凭证申请：`api/transfer/certificate/apply`
- 转账凭证查询：`api/transfer/certificate/query`
- 资金转账支付：`api/fund/trans/pay`

在线文档：
- 转账余额查询：https://doc.jeequan.com/#/integrate/jqf/api/51
- 转账凭证申请：https://doc.jeequan.com/#/integrate/jqf/api/52
- 转账凭证查询：https://doc.jeequan.com/#/integrate/jqf/api/53
- 资金转账支付：https://doc.jeequan.com/#/integrate/jqf/api/54

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
转账扩展示例
包含：余额查询、凭证申请、凭证查询、资金转账支付
"""

from jeepay import AppConfig, Transfer

# ============================================================
# 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")


# ============================================================
# 1. 转账余额查询
# ============================================================
result = Transfer.balance_query(
    ifCode="aliaqfpay"  # 支付接口代码
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"可用余额: {data.get('balanceAmount')} 分")
else:
    print(f"余额查询失败: {result.get('msg')}")


# ============================================================
# 2. 转账凭证申请
# ============================================================
result = Transfer.certificate_apply(
    transferId="T1862403098937430017",  # 转账订单号（与 mchOrderNo 二选一）
    # mchOrderNo="T20260521160000",      # 商户转账单号
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"凭证申请成功，转账订单号: {data.get('transferId')}")
else:
    print(f"凭证申请失败: {result.get('msg')}")


# ============================================================
# 3. 转账凭证查询（下载）
# ============================================================
result = Transfer.certificate_query(
    transferId="T1862403098937430017",  # 转账订单号（与 mchOrderNo 二选一）
    # mchOrderNo="T20260521160000",      # 商户转账单号
    fileId="2022211111"                  # 文件ID
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"下载地址: {data.get('downloadUrl')}")
else:
    print(f"凭证查询失败: {result.get('msg')}")


# ============================================================
# 4. 资金转账支付（充值）
# ============================================================
result = Transfer.trans_pay(
    ifCode="aliaqfpay",     # 支付接口代码
    amount=1,               # 充值金额，单位：分
    rechargeType=0          # 充值类型：0-线上充值 1-线下充值
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"充值类型: {data.get('rechargeType')}")
    if data.get("rechargeType") == 0:
        # 线上充值返回表单数据
        print(f"表单内容: {data.get('formContent')}")
    else:
        # 线下充值返回银行卡信息
        print(f"卡号: {data.get('cardNo')}")
        print(f"开户行: {data.get('cardBank')}")
else:
    print(f"充值失败: {result.get('msg')}")
```
