# 提现

API路径：

- 提现余额查询：`api/cashout/balance/query`
- 创建提现订单：`api/cashout/order/create`
- 查询提现订单：`api/cashout/order/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/77

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
提现示例
包含：提现余额查询、创建提现订单、查询提现订单。
"""

import time
from jeepay import AppConfig, Cashout

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 提现余额查询
# ============================================================
balance_result = Cashout.balance_query(
    ifCode="dgpay"  # 支付接口代码
)

if balance_result.get("code") == 0:
    balance_data = balance_result.get("data", {})
    print(f"当前余额: {balance_data.get('balance')} 分")
    print(f"冻结余额: {balance_data.get('frozenBalance')} 分")
    print(f"总余额: {balance_data.get('totalBalance')} 分")
    print(f"渠道账户余额详情: {balance_data.get('channelBalanceDetail')}")
else:
    print(f"余额查询失败: {balance_result.get('msg')}")
    # 可以选择终止流程或继续

# ============================================================
# 3. 创建提现订单
# ============================================================
create_result = Cashout.order_create(
    mchOrderNo=f"CO{round(time.time() * 1000)}",  # 商户提现单号（保证唯一）
    amount=1,                                       # 提现金额，单位：分
    ifCode="dgpay",                                 # 支付接口代码
    notifyUrl="https://your-domain.com/notify/cashout",  # 异步通知地址
    # channelExtra="",                              # 特定渠道额外参数（可选）
    remark="测试提现备注"                            # 备注（可选）
)

if create_result.get("code") == 0:
    print("提现申请成功！")
    create_data = create_result.get("data", {})
    print(f"提现订单号: {create_data.get('cashoutOrderId')}")
    print(f"商户单号: {create_data.get('mchOrderNo')}")
    print(f"提现状态: {create_data.get('state')}")
    # state 状态说明：
    #   0 - 提现单创建
    #   1 - 提现中
    #   2 - 提现成功
    #   3 - 提现失败
else:
    print(f"提现申请失败！code={create_result.get('code')}, msg={create_result.get('msg')}")

# ============================================================
# 4. 查询提现订单
# ============================================================
query_result = Cashout.order_query(
    mchOrderNo="CO1716700800000",  # 商户提现单号（与cashoutOrderId二选一）
    # cashoutOrderId="C2026052600001001",  # 支付系统提现订单号
)

if query_result.get("code") == 0:
    query_data = query_result.get("data", {})
    print(f"提现订单号: {query_data.get('cashoutOrderId')}")
    print(f"商户单号: {query_data.get('mchOrderNo')}")
    print(f"支付接口: {query_data.get('ifCode')}")
    print(f"提现金额: {query_data.get('amount')} 分")
    print(f"提现前余额: {query_data.get('balance')} 分")
    print(f"提现状态: {query_data.get('state')}")
    print(f"渠道提现单号: {query_data.get('channelOrderNo')}")
    print(f"提现成功时间: {query_data.get('successTime')}")
    print(f"创建时间: {query_data.get('createdAt')}")
    print(f"提现银行: {query_data.get('bankName')}")
    print(f"提现账户: {query_data.get('accountNo')}")
    print(f"账户名称: {query_data.get('accountName')}")
    print(f"备注: {query_data.get('remark')}")
else:
    print(f"查询失败！code={query_result.get('code')}, msg={query_result.get('msg')}")
```
