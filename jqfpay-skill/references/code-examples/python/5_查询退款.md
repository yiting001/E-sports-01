# 查询退款

API路径：`api/refund/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/49

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
查询退款示例
通过商户退款单号或退款订单号查询退款状态。
"""

from jeepay import AppConfig, Pay

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 通过商户退款单号查询
# ============================================================
result = Pay.refund_query(
    mchRefundNo="R20260521150000",  # 商户退款单号（与 refundOrderId 二选一）
)

# ============================================================
# 3. 通过退款订单号查询
# ============================================================
# result = Pay.refund_query(
#     refundOrderId="RF202605211500000001",  # 退款订单号（与 mchRefundNo 二选一）
# )

# ============================================================
# 4. 处理返回结果
# ============================================================
if result.get("code") == 0:
    print("查询成功！")
    refund_data = result.get("data", {})
    print(f"退款订单号: {refund_data.get('refundOrderId')}")
    print(f"商户退款单号: {refund_data.get('mchRefundNo')}")
    print(f"原支付订单号: {refund_data.get('payOrderId')}")
    print(f"退款金额: {refund_data.get('refundAmount')} 分")
    print(f"退款状态: {refund_data.get('state')}")
    # state 状态说明：
    #   0 - 退款生成
    #   1 - 退款中
    #   2 - 退款成功
    #   3 - 退款失败
    print(f"退款原因: {refund_data.get('refundReason')}")
    print(f"渠道订单号: {refund_data.get('channelOrderNo')}")
    print(f"成功退款时间: {refund_data.get('successTime')}")
else:
    print(f"查询失败！code={result.get('code')}, msg={result.get('msg')}")
```
