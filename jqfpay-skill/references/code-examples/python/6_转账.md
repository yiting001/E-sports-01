# 转账

API路径：`api/transferOrder`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/45

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
转账示例
将商户余额转账到指定账户（微信/支付宝/银行卡）。
"""

from jeepay import AppConfig, Transfer

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 查询可转账余额（建议先查询余额再发起转账）
# ============================================================
balance_result = Transfer.balance_query()
if balance_result.get("code") == 0:
    balance_data = balance_result.get("data", {})
    print(f"可转账余额: {balance_data} 分")
else:
    print(f"余额查询失败: {balance_result.get('msg')}")
    # 可以选择终止流程或继续

# ============================================================
# 3. 发起转账
# ============================================================
result = Transfer.create(
    mchOrderNo="T20260521160000",        # 商户转账单号（需保证唯一）
    entryType="WX_CASH",                 # 转账入口类型
    # WX_CASH - 微信零钱
    # ALI_CASH - 支付宝余额
    # BANK_CARD - 银行卡

    amount=100,                           # 转账金额，单位：分（100 = 1元）
    currency="cny",                       # 币种
    subject="测试转账",                   # 转账标题
    transferDesc="用户提现",              # 转账描述
    notifyUrl="https://your-domain.com/api/notify/transfer",  # 异步通知地址

    # 收款方信息（根据 entryType 不同，参数不同）
    # 微信零钱转账
    accountNo="o6BcIwvSiRpfS8e_UyfQNrYuk2LI",  # 收款人OpenID
    accountName="张三",                          # 收款人姓名（可选，需与微信实名一致）

    # 支付宝余额转账
    # accountNo="2088702585070844",         # 收款人支付宝UID
    # accountName="张三",

    # 银行卡转账
    # accountNo="6222021234567890123",     # 收款银行卡号
    # accountName="张三",                   # 收款人姓名（必填）
    # bankName="工商银行",                  # 银行名称
    # bankBranchName="北京分行",            # 支行名称（可选）

    # 收款方附加信息（可选）
    # receiverExtInfo='{"contactEmail":"test@example.com"}',
)

# ============================================================
# 4. 处理返回结果
# ============================================================
if result.get("code") == 0:
    print("转账申请成功！")
    transfer_data = result.get("data", {})
    print(f"转账订单号: {transfer_data.get('transferId')}")
    print(f"商户转账单号: {transfer_data.get('mchOrderNo')}")
    print(f"转账状态: {transfer_data.get('state')}")
    # state 状态说明：
    #   0 - 转账生成
    #   1 - 转账中
    #   2 - 转账成功
    #   3 - 转账失败
    print(f"转账金额: {transfer_data.get('amount')} 分")
else:
    print(f"转账申请失败！code={result.get('code')}, msg={result.get('msg')}")

# ============================================================
# 5. 查询转账状态
# ============================================================
query_result = Transfer.transfer_query(
    mchOrderNo="T20260521160000",  # 商户转账单号（与 transferId 二选一）
    # transferId="TF202605211600000001",  # 转账订单号
)
if query_result.get("code") == 0:
    query_data = query_result.get("data", {})
    print(f"转账状态: {query_data.get('state')}")
    print(f"成功时间: {query_data.get('successTime')}")
```
