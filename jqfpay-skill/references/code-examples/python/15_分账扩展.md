# 分账扩展

分账扩展接口包含：绑定/查询分账用户、查询/完结分账、余额分账执行/查询、渠道余额查询/提现、提现记录查询、结算更新。

API路径：
- 绑定分账用户：`api/division/receiver/bind`
- 查询分账绑定：`api/division/receiver/bindQuery`
- 查询分账：`api/division/query`
- 完成分账：`api/division/finish`
- 余额分账执行：`api/division/balance/exec`
- 余额分账查询：`api/division/balance/query`
- 渠道余额查询：`api/division/receiver/channelBalanceQuery`
- 渠道余额提现：`api/division/receiver/channelBalanceCashout`
- 提现记录查询：`api/division/receiver/cashoutRecordQuery`
- 结算更新：`api/division/receiver/settle/update`

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
分账扩展示例
包含：绑定/查询用户、查询/完结分账、余额分账、渠道余额、提现记录、结算更新
"""

import time
from jeepay import AppConfig, Division

# ============================================================
# 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")


# ============================================================
# 1. 绑定分账用户
# ============================================================
result = Division.bind_user(
    ifCode="demomockpay",              # 接口代码
    mchReceiverId="23472389482349",    # 商户分账用户ID
    receiverAlias="15521254124",       # 接收者账号别名
    accType=0,                         # 账号类型：0-个人(对私) 1-商户(对公)
    accNo="15521254124",               # 分账接收账号
    relationType="PARTNER",            # 分账关系类型
    divisionProfit=0.3                 # 默认分账比例（0.3 = 30%）
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"绑定成功，接收者ID: {data.get('receiverId')}, 绑定状态: {data.get('bindState')}")
else:
    print(f"绑定失败: {result.get('msg')}")


# ============================================================
# 2. 查询分账绑定
# ============================================================
result = Division.bind_query(
    receiverId=800093                  # 分账接收者ID（与 mchReceiverId 二选一）
    # mchReceiverId="23472389482349",  # 商户分账用户ID
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"绑定状态: {data.get('bindState')}, 分账比例: {data.get('divisionProfit')}")


# ============================================================
# 3. 查询分账
# ============================================================
result = Division.exec_query(
    payOrderId="P1955471412902871041",     # 支付订单号（与 mchOrderNo 二选一）
    batchOrderId="D1955471507933216769"    # 系统分账批次号
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"分账状态: {data.get('divisionState')}")  # 0-未分账 1-待分账 2-分账中 3-已分账 4-完结中 5-已完结


# ============================================================
# 4. 完成分账
# ============================================================
result = Division.exec_finish(
    payOrderId="P1955471412902871041",     # 支付订单号（与 mchOrderNo 二选一）
    # mchOrderNo="20160427210604000490",   # 商户单号
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"完结状态: {data.get('state')}")  # 1-成功 2-失败 3-处理中


# ============================================================
# 5. 余额分账执行
# ============================================================
receivers = [
    {
        "receiverId": 800085,
        "divisionAmount": "1"
    }
]
result = Division.balance_exec(
    mchDivisionNo=f"mho{round(time.time() * 1000)}",  # 商户分账单号
    ifCode="demomockpay",              # 支付接口代码
    totalAmount=1,                     # 分账总金额，单位：分
    receivers=str(receivers)           # 分账接收者账号列表
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"分账状态: {data.get('state')}")  # 1-成功 2-失败 3-待分账 4-分账中
    print(f"系统批次号: {data.get('batchOrderId')}")


# ============================================================
# 6. 余额分账查询
# ============================================================
result = Division.balance_query(
    mchDivisionNo="mho1716288000000",     # 商户分账单号（与 batchOrderId 二选一）
    # batchOrderId="D202605211700000001",  # 系统分账批次号
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"系统批次号: {data.get('batchOrderId')}")
    print(f"详细信息: {data.get('records')}")


# ============================================================
# 7. 渠道余额查询
# ============================================================
result = Division.channel_balance_query(
    receiverId=800057                  # 分账用户ID
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"可提现余额: {data.get('balanceAmount')} 分")
    print(f"不可提现余额: {data.get('unBalanceAmount')} 分")


# ============================================================
# 8. 渠道余额提现
# ============================================================
result = Division.channel_balance_cashout(
    receiverId=800057,                 # 分账用户ID
    mchOrderNo=f"CO{round(time.time() * 1000)}",  # 商户提现单号
    cashoutAmount=100,                 # 提现金额，单位：分
    remark="分账提现"                   # 备注
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"提现记录单号: {data.get('rid')}")
    print(f"发起状态: {data.get('state')}")        # 1-成功 0-失败
    print(f"提现状态: {data.get('cashoutState')}")  # 0-创建 1-提现中 2-成功 3-失败


# ============================================================
# 9. 提现记录查询
# ============================================================
result = Division.cashout_record_query(
    receiverId=800057,                 # 分账用户ID
    pageNumber=1,                      # 分页页码，默认1
    pageSize=20                        # 分页条数，默认20
)
if result.get("code") == 0:
    data = result.get("data", {})
    print(f"总条数: {data.get('total')}, 是否有下一页: {data.get('hasNext')}")


# ============================================================
# 10. 结算更新
# ============================================================
result = Division.receiver_settle_update(
    receiverId=800046,                              # 分账接收者ID
    settAccountName="测试",                          # 结算账号名称
    settAccountNo="6217375892834829482",            # 结算账号
    settAccountBankName="中国建设银行",              # 结算银行名称
    settAccountBankBranchCode="105143000017",       # 结算银行联行号
    settAccountBankBranchName="",                   # 结算银行支行名称
    channelExtInfo=""                                # 渠道特殊信息
)
if result.get("code") == 0:
    print("结算信息更新成功")
```
