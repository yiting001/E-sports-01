# 发票

API路径：

- 开票：`api/invoice/open`
- 查询发票：`api/invoice/query`
- 电子发票URL：`api/invoice/elecUrl`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/74

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
发票示例
包含：开票、查询发票、获取电子发票URL。
"""

import time
from jeepay import AppConfig, Invoice

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 开票
# ============================================================
buyer_info = {
    "buyerName": "测试公司",
    "taxNo": "91110108MA01XXXXXX",
    "pushPhone": "13800138000",
}

open_result = Invoice.open_invoice(
    mchOrderNo=f"FP{round(time.time() * 1000)}",  # 商户开票单号（保证唯一）
    invoiceWay=2,           # 开票方式: 2-获取开票链接
    amount=100,             # 含税开票金额，单位：分（100 = 1元）
    clientIp="127.0.0.1",  # 客户端IP地址
    buyerInfo=str(buyer_info),  # 购方信息
    # notifyUrl="https://your-domain.com/notify/invoice",  # 异步通知地址（可选）
    # expiredDay=7,          # 开票链接有效期（可选）
    # remark="备注信息",     # 备注（可选）
)

if open_result.get("code") == 0:
    print("开票申请成功！")
    open_data = open_result.get("data", {})
    print(f"发票单号: {open_data.get('invoiceOrderId')}")
    print(f"开票状态: {open_data.get('state')}")
    # state 状态说明：
    #   0 - 订单生成
    #   1 - 开票中
    #   2 - 开票成功
    #   3 - 开票失败
    #   4 - 开票任务关闭
    #   5 - 已冲红
    #   6 - 已作废
    print(f"开票链接: {open_data.get('channelInvoiceQrcodeUrl')}")
    print(f"链接失效时间: {open_data.get('expiredTime')}")
else:
    print(f"开票申请失败！code={open_result.get('code')}, msg={open_result.get('msg')}")

# ============================================================
# 3. 查询发票
# ============================================================
query_result = Invoice.query_invoice(
    invoiceOrderId="FP2026052600001001",  # 发票单号（与mchOrderNo二选一）
    # mchOrderNo="FP1716700800000",       # 商户开票单号
)

if query_result.get("code") == 0:
    query_data = query_result.get("data", {})
    print(f"发票单号: {query_data.get('invoiceOrderId')}")
    print(f"商户单号: {query_data.get('mchOrderNo')}")
    print(f"开票金额: {query_data.get('amount')} 分")
    print(f"税额: {query_data.get('feeAmount')} 分")
    print(f"开票状态: {query_data.get('state')}")
    print(f"渠道发票单号: {query_data.get('channelInvoiceOrderNo')}")
    print(f"购方信息: {query_data.get('buyerInfo')}")
    print(f"发票信息: {query_data.get('invoiceInfo')}")
    print(f"备注: {query_data.get('remark')}")
else:
    print(f"查询失败！code={query_result.get('code')}, msg={query_result.get('msg')}")

# ============================================================
# 4. 获取电子发票URL（全电账号管理地址）
# ============================================================
elec_url_result = Invoice.elec_url_invoice()

if elec_url_result.get("code") == 0:
    elec_data = elec_url_result.get("data", {})
    print(f"全电账号管理URL: {elec_data.get('etaxUserUrl')}")
else:
    print(f"获取失败！code={elec_url_result.get('code')}, msg={elec_url_result.get('msg')}")
```
