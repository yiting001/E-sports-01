# 协议支付

API路径：`api/pay/bankAgreePreBind` + `api/pay/bankAgreeConfirmBind` + `api/pay/bankAgreeProtocolpayConfirm`

在线文档：
- 协议预绑定：https://doc.jeequan.com/#/integrate/jqf/api/131
- 协议确认绑定：https://doc.jeequan.com/#/integrate/jqf/api/132
- 协议支付确认：https://doc.jeequan.com/#/integrate/jqf/api/133

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
协议支付完整流程示例
流程：预绑定 -> 确认绑定 -> 协议支付 -> （可选）重发短信 / 解绑
"""

from jeepay import AppConfig, AgreePay, Pay

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 步骤一：协议预绑定银行卡
# ============================================================
pre_bind_result = AgreePay.bank_pre_bind(
    userName="张三",                       # 持卡人姓名
    idType="ID_CARD",                      # 证件类型：ID_CARD-身份证
    idNo="110101199001011234",             # 证件号码
    cardNo="6222021234567890123",          # 银行卡号
    bankName="工商银行",                   # 银行名称
    phone="18600001111",                   # 预留手机号
    notifyUrl="https://your-domain.com/api/notify/agree_bind",  # 绑定结果通知地址
)

if pre_bind_result.get("code") == 0:
    pre_bind_data = pre_bind_result.get("data", {})
    print("预绑定成功！")
    bind_token = pre_bind_data.get("bindToken")
    print(f"绑定Token: {bind_token}")
    print(f"短信已发送到预留手机号")
else:
    print(f"预绑定失败！code={pre_bind_result.get('code')}, msg={pre_bind_result.get('msg')}")


# ============================================================
# 步骤二：确认绑定（用户输入短信验证码后）
# ============================================================
confirm_bind_result = AgreePay.bank_confirm_bind(
    bindToken=bind_token,                  # 预绑定返回的Token
    smsCode="123456",                      # 用户输入的短信验证码
)

if confirm_bind_result.get("code") == 0:
    confirm_data = confirm_bind_result.get("data", {})
    print("绑定成功！")
    agreement_id = confirm_data.get("agreementId")
    print(f"协议ID: {agreement_id}")
    print(f"银行卡尾号: {confirm_data.get('cardLast4')}")
else:
    print(f"确认绑定失败！code={confirm_bind_result.get('code')}, msg={confirm_bind_result.get('msg')}")


# ============================================================
# 步骤三：使用协议支付（通过统一支付下单）
# ============================================================
pay_result = Pay.create(
    mchOrderNo="M20260521180000",
    wayCode="BANK_AGREE",                  # 协议支付方式
    amount=100,
    subject="协议支付测试",
    notifyUrl="https://your-domain.com/api/notify/pay",
    channelExtra='{'
    '  "agreementId": "' + agreement_id + '",'
    '  "smsCodeType": "01"'
    '}',
)

if pay_result.get("code") == 0:
    pay_data = pay_result.get("data", {})
    print("支付请求已发送！")
    print(f"支付订单号: {pay_data.get('payOrderId')}")
    print(f"订单状态: {pay_data.get('state')}")
else:
    print(f"支付请求失败！code={pay_result.get('code')}, msg={pay_result.get('msg')}")


# ============================================================
# 步骤四：确认协议支付（如需短信验证）
# ============================================================
confirm_pay_result = AgreePay.bank_confirm_pay(
    payOrderId="P202605211800000001",     # 支付订单号
    smsCode="654321",                      # 用户输入的支付验证码
)

if confirm_pay_result.get("code") == 0:
    print("支付确认成功！")
    print(f"确认结果: {confirm_pay_result.get('data')}")
else:
    print(f"支付确认失败！code={confirm_pay_result.get('code')}, msg={confirm_pay_result.get('msg')}")


# ============================================================
# 可选：重发支付短信
# ============================================================
resend_result = AgreePay.bank_resend_pay_sms(
    payOrderId="P202605211800000001",
)
if resend_result.get("code") == 0:
    print("短信已重新发送！")


# ============================================================
# 可选：解绑协议
# ============================================================
unbind_result = AgreePay.bank_unbind(
    agreementId=agreement_id,
)
if unbind_result.get("code") == 0:
    print("解绑成功！")


# ============================================================
# 可选：查询支持的银行列表
# ============================================================
bank_list_result = AgreePay.bank_list_query()
if bank_list_result.get("code") == 0:
    banks = bank_list_result.get("data", [])
    print(f"支持的银行列表: {banks}")
```
