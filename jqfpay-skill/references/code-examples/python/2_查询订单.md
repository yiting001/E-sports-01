# 查询订单

API路径：`api/pay/query`

在线文档：https://doc.jeequan.com/#/integrate/jqf/api/40

## SDK调用示例（Python）

```python
# -*- coding: utf-8 -*-
"""
查询支付订单示例
支持通过商户订单号或支付订单号查询。
"""

from jeepay import AppConfig, Pay

# ============================================================
# 1. 初始化 SDK 配置
# ============================================================
AppConfig.set_mch_no("M1671695486")
AppConfig.set_app_id("607cc0a0ee11e11e7c80")
AppConfig.set_api_key("YourApiKey1234567890")

# ============================================================
# 2. 通过商户订单号查询
# ============================================================
result = Pay.pay_query(
    mchOrderNo="M20260521143000",  # 商户订单号（与 payOrderId 二选一）
)

# ============================================================
# 3. 通过支付订单号查询
# ============================================================
# result = Pay.pay_query(
#     payOrderId="P202605211430000001",  # 支付订单号（与 mchOrderNo 二选一）
# )

# ============================================================
# 4. 处理返回结果
# ============================================================
if result.get("code") == 0:
    print("查询成功！")
    order_data = result.get("data", {})
    print(f"支付订单号: {order_data.get('payOrderId')}")
    print(f"商户订单号: {order_data.get('mchOrderNo')}")
    print(f"订单金额: {order_data.get('orderAmount')} 分")
    print(f"订单状态: {order_data.get('state')}")
    # state 状态说明：
    #   0 - 订单生成（未支付）
    #   1 - 支付中
    #   2 - 支付成功
    #   3 - 支付失败
    #   4 - 已撤销
    #   5 - 已退款
    #   6 - 已关闭
    print(f"支付渠道: {order_data.get('wayCode')}")
    print(f"成功支付时间: {order_data.get('successTime')}")
    print(f"渠道订单号: {order_data.get('channelOrderNo')}")
else:
    print(f"查询失败！code={result.get('code')}, msg={result.get('msg')}")
```
